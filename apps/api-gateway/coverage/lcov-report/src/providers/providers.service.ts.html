      (r) => r.is_active !== false && !r.deleted_at,
    );

    let stated = 0;
    for (const r of live) {
      const code = (r.usual_currency ?? "").trim().toUpperCase();
      if (code !== "" && isIso4217(code)) {

      unstated.push({
        id: r.id,
        name: (r.name ?? "").trim() || "This vendor",
