```python
# =============================================================================
# COST-03: Per-restaurant cap check
# =============================================================================
AUTO_BLOCK_THRESHOLD = 0.3  # QUAL-01 / D-05: completeness < 0.3 → auto_blocked
PER_RESTAURANT_CAP_USD = 2.00  # D-03
COST_PER_PAGE_ESTIMATE = 0.05  # $0.05/page conservative upper bound (D-03)


def _get_month_start() -> str:
    now = datetime.now(timezone.utc)
    return datetime(now.year, now.month, 1, tzinfo=timezone.utc).isoformat()


async def _send_cap_alert_email(restaurant_id: str, existing_spend: float, estimated_cost: float) -> None:
    """Fire-and-forget alert email when per-restaurant cap is breached."""
    try:
        from config.settings import get_settings
        from services.email_client import EmailClient
        settings = get_settings()
        if not (settings.manager_email and settings.gmail_user and settings.gmail_password):
            logger.warning("Cap alert: MANAGER_EMAIL/GMAIL_USER/GMAIL_PASSWORD not configured")
            return
        email_client = EmailClient(
            backend="gmail",
            gmail_user=settings.gmail_user,
            gmail_password=settings.gmail_password,
        )
        await email_client.send_email(
            to=settings.manager_email,
            subject=f"[WineOps] Per-restaurant cap exceeded: restaurant_id={restaurant_id}",
            body=(
                f"Per-restaurant extraction cap exceeded.\n\n"
                f"Restaurant ID: {restaurant_id}\n"
                f"Current month spend: ${existing_spend:.4f}\n"
                f"Estimated this request: ${estimated_cost:.4f}\n"
                f"Cap: ${PER_RESTAURANT_CAP_USD:.2f}\n\n"
                f"Extraction was rejected with HTTP 402."
            ),
        )
    except Exception as exc:
        logger.warning(f"Failed to send cap alert email: {exc}")


def _preflight_cap_check(supabase, restaurant_id: str, page_count: int) -> None:
    """
    Raises HTTPException(402) if this restaurant would breach the $2.00 monthly cap.
    Must be called BEFORE extractor.extract_menu() (D-03: no mid-extraction abort).
    """
    if supabase is None:
        return  # Supabase not configured — skip cap check (dev/test environments)
    estimated_cost = page_count * COST_PER_PAGE_ESTIMATE
    try:
        resp = (
            supabase.table("api_spend")
            .select("cost_usd")
            .eq("restaurant_id", restaurant_id)
            .gte("timestamp", _get_month_start())
            .execute()
        )
        existing_spend = sum(r["cost_usd"] for r in (resp.data or []))
    except Exception as exc:
        logger.warning(f"Pre-flight cap check failed (non-fatal — allowing extraction): {exc}")
        return  # fail open: if the check errors, allow extraction to proceed
    if existing_spend + estimated_cost > PER_RESTAURANT_CAP_USD:
        asyncio.create_task(
            _send_cap_alert_email(restaurant_id, existing_spend, estimated_cost)
        )
        raise HTTPException(
            status_code=402,
            detail=(
                f"Per-restaurant cap exceeded: ${existing_spend:.4f} spent this month + "
                f"${estimated_cost:.4f} estimated > ${PER_RESTAURANT_CAP_USD:.2f} cap"
            ),
        )
```
