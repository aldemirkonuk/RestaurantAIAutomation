import {
  BadRequestException,
  ConflictException,
  ForbiddenException,
  Injectable,
  InternalServerErrorException,
  Logger,
  NotFoundException,
} from '@nestjs/common';
import { DatabaseService } from '../database/database.service';

@Injectable()
export class MembersService {
  private readonly logger = new Logger(MembersService.name);

  constructor(private readonly databaseService: DatabaseService) {}

  /**
   * Verify the acting user has an active URA row for the restaurant.
   * Throws ForbiddenException if not a member.
   * Throws ForbiddenException if requiredRole is set and caller's role does not match.
   */
  private async assertMembership(
    actorUserId: string,
    restaurantId: string,
    requiredRole?: 'owner' | 'manager' | 'owner|manager',
  ): Promise<{ role: string }> {
    const { data: access } = await this.databaseService.supabase
      .from('user_restaurant_access')
      .select('role')
      .eq('user_id', actorUserId)
      .eq('restaurant_id', restaurantId)
      .eq('is_active', true)
      .maybeSingle();

    if (!access) throw new ForbiddenException('Access denied to this restaurant');

    if (requiredRole === 'owner' && access.role !== 'owner') {
      throw new ForbiddenException('Only owners can perform this action');
    }
    if (requiredRole === 'owner|manager' && access.role === 'staff') {
      throw new ForbiddenException('Only owners and managers can perform this action');
    }

    return access;
  }

  /**
   * MEMBER-03: Return all active members of a specific restaurant.
   * Joins users table for profile info and organization_invites for invite metadata.
   */
  async getMembers(actorUserId: string, restaurantId: string): Promise<any[]> {
    await this.assertMembership(actorUserId, restaurantId);

    const { data: members, error } = await this.databaseService.supabase
      .from('user_restaurant_access')
      .select(`
        id,
        role,
        granted_at,
        valid_until,
        is_active,
        users!user_id ( user_id, name, email, avatar_url, auth_provider ),
        organization_invites!invited_via ( code, invited_by )
      `)
      .eq('restaurant_id', restaurantId)
      .eq('is_active', true)
      .order('granted_at', { ascending: true });

    if (error) {
      this.logger.error(`getMembers failed for restaurant ${restaurantId}: ${error.message}`);
      return [];
    }

    return members ?? [];
  }

  /**
   * Return pending (not-yet-used, not-expired) invites for the Team tab pending section.
   * Only owners and managers can see pending invites (CONTEXT D-07).
   */
  async getInvites(actorUserId: string, restaurantId: string): Promise<any[]> {
    await this.assertMembership(actorUserId, restaurantId, 'owner|manager');

    const { data: invites, error } = await this.databaseService.supabase
      .from('organization_invites')
      .select('id, code, role, expires_at, created_at, users!invited_by ( name )')
      .eq('restaurant_id', restaurantId)
      .is('used_at', null)
      .gt('expires_at', new Date().toISOString())
      .order('created_at', { ascending: false });

    if (error) {
      this.logger.error(`getInvites failed for restaurant ${restaurantId}: ${error.message}`);
      return [];
    }

    return invites ?? [];
  }

  /**
   * MEMBER-04: Update a member's role atomically.
   * Updates user_restaurant_access.role AND users.role for backward compat.
   * Only owners can change roles (CONTEXT D-07, D-13).
   */
  async updateMemberRole(
    actorUserId: string,
    restaurantId: string,
    targetUserId: string,
    newRole: 'owner' | 'manager' | 'staff',
  ): Promise<void> {
    await this.assertMembership(actorUserId, restaurantId, 'owner');

    // Last-owner protection: owner demoting themselves — block if only owner
    if (actorUserId === targetUserId && newRole !== 'owner') {
      const { count } = await this.databaseService.supabase
        .from('user_restaurant_access')
        .select('*', { count: 'exact', head: true })
        .eq('restaurant_id', restaurantId)
        .eq('role', 'owner')
        .eq('is_active', true);

      if ((count ?? 0) <= 1) {
        throw new BadRequestException(
          "You're the only owner. Transfer ownership or delete the restaurant first.",
        );
      }
    }

    // Update user_restaurant_access.role (authoritative)
    const { error: uraErr } = await this.databaseService.supabase
      .from('user_restaurant_access')
      .update({ role: newRole })
      .eq('user_id', targetUserId)
      .eq('restaurant_id', restaurantId);

    if (uraErr) {
      this.logger.error(`updateMemberRole URA update failed: ${uraErr.message}`);
      throw new InternalServerErrorException('Failed to update member role');
    }

    // Also update users.role for backward compat (legacy column — kept until Phase 34+)
    await this.databaseService.supabase
      .from('users')
      .update({ role: newRole })
      .eq('user_id', targetUserId);
  }

  /**
   * MEMBER-05: Hard-delete the URA row.
   * User account and all historical data preserved — they simply lose access.
   * Per CONTEXT D-15: hard delete (not soft deactivation).
   * Per CONTEXT D-18: blocked if removing the last owner.
   */
  async removeMember(
    actorUserId: string,
    restaurantId: string,
    targetUserId: string,
  ): Promise<void> {
    await this.assertMembership(actorUserId, restaurantId, 'owner');

    // Fetch target's current role to check last-owner guard
    const { data: targetAccess } = await this.databaseService.supabase
      .from('user_restaurant_access')
      .select('role')
      .eq('user_id', targetUserId)
      .eq('restaurant_id', restaurantId)
      .eq('is_active', true)
      .maybeSingle();

    if (!targetAccess) throw new NotFoundException('Member not found in this restaurant');

    if (targetAccess.role === 'owner') {
      // Count active owners — block if this is the last one (CONTEXT D-18)
      const { count } = await this.databaseService.supabase
        .from('user_restaurant_access')
        .select('*', { count: 'exact', head: true })
        .eq('restaurant_id', restaurantId)
        .eq('role', 'owner')
        .eq('is_active', true);

      if ((count ?? 0) <= 1) {
        throw new BadRequestException(
          "You're the only owner. Transfer ownership or delete the restaurant first.",
        );
      }
    }

    // Hard delete URA row (CONTEXT D-15)
    const { error } = await this.databaseService.supabase
      .from('user_restaurant_access')
      .delete()
      .eq('user_id', targetUserId)
      .eq('restaurant_id', restaurantId);

    if (error) {
      this.logger.error(`removeMember delete failed: ${error.message}`);
      throw new InternalServerErrorException('Failed to remove member');
    }
  }

  /**
   * MEMBER-10: Manager-initiated add of existing user (by email) to a restaurant.
   * Enables adding OAuth users who have an account but no password.
   * No invite code required — manager directly grants access.
   */
  async addMember(
    actorUserId: string,
    restaurantId: string,
    email: string,
    role: 'owner' | 'manager' | 'staff',
  ): Promise<void> {
    const actorAccess = await this.assertMembership(actorUserId, restaurantId, 'owner|manager');

    // Managers can only add staff (CONTEXT D-07)
    if (actorAccess.role === 'manager' && role !== 'staff') {
      throw new ForbiddenException('Managers can only add staff members');
    }

    // Look up user by email
    const { data: targetUser } = await this.databaseService.supabase
      .from('users')
      .select('user_id')
      .eq('email', email)
      .maybeSingle();

    if (!targetUser) {
      throw new NotFoundException(
        'User not found. Send them an invite link to create an account first.',
      );
    }

    // Check for existing membership
    const { data: existingAccess } = await this.databaseService.supabase
      .from('user_restaurant_access')
      .select('id')
      .eq('user_id', targetUser.user_id)
      .eq('restaurant_id', restaurantId)
      .maybeSingle();

    if (existingAccess) {
      throw new ConflictException('User is already a member of this restaurant');
    }

    // Get restaurant's organization for organization_members upsert
    const { data: restaurant } = await this.databaseService.supabase
      .from('restaurants')
      .select('organization_id')
      .eq('id', restaurantId)
      .maybeSingle();

    // Insert URA row (works for OAuth users — no password involved)
    const { error: uraErr } = await this.databaseService.supabase
      .from('user_restaurant_access')
      .insert({
        user_id: targetUser.user_id,
        restaurant_id: restaurantId,
        role,
        invited_via: null,
        is_active: true,
      });

    if (uraErr) {
      this.logger.error(`addMember URA insert failed: ${uraErr.message}`);
      throw new InternalServerErrorException('Failed to add member');
    }

    // Upsert organization_members so user appears in org-level queries
    if (restaurant?.organization_id) {
      await this.databaseService.supabase.from('organization_members').upsert(
        { organization_id: restaurant.organization_id, user_id: targetUser.user_id, role },
        { onConflict: 'organization_id,user_id' },
      );
    }
  }

  /**
   * Revoke a pending invite by code (owner or manager).
   * DELETE from organization_invites where code matches and restaurant matches.
   */
  async revokeInvite(
    actorUserId: string,
    restaurantId: string,
    code: string,
  ): Promise<void> {
    await this.assertMembership(actorUserId, restaurantId, 'owner|manager');

    const { error } = await this.databaseService.supabase
      .from('organization_invites')
      .delete()
      .eq('code', code.toUpperCase())
      .eq('restaurant_id', restaurantId)
      .is('used_at', null);

    if (error) {
      this.logger.error(`revokeInvite failed: ${error.message}`);
      throw new InternalServerErrorException('Failed to revoke invite');
    }
  }
}

import {
  Body,
  Controller,
  Delete,
  ForbiddenException,
  Get,
  Logger,
  Param,
  Patch,
  Post,
  Req,
  UseGuards,
  UsePipes,
  ValidationPipe,
} from '@nestjs/common';
import { MembersService } from './members.service';
import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
import { UpdateMemberRoleDto } from '../auth/dto/update-member-role.dto';
import { Request } from 'express';

interface AuthenticatedUser {
  userId: string;
  role: string;
}

@Controller('restaurants')
@UseGuards(JwtAuthGuard)
@UsePipes(new ValidationPipe({ whitelist: true }))
export class MembersController {
  private readonly logger = new Logger(MembersController.name);

  constructor(private readonly membersService: MembersService) {}

  /**
   * MEMBER-03: Get all active members of a specific restaurant.
   * Accessible to all members of that restaurant (owner/manager/staff).
   */
  @Get(':restaurantId/members')
  async getMembers(
    @Req() req: Request & { user: AuthenticatedUser },
    @Param('restaurantId') restaurantId: string,
  ) {
    const userId = (req.user as AuthenticatedUser)?.userId;
    if (!userId) throw new ForbiddenException('Missing user identity');
    return this.membersService.getMembers(userId, restaurantId);
  }

  /**
   * Get pending invites for the restaurant's Team tab.
   * Only owners and managers can see pending invites.
   */
  @Get(':restaurantId/invites')
  async getInvites(
    @Req() req: Request & { user: AuthenticatedUser },
    @Param('restaurantId') restaurantId: string,
  ) {
    const userId = (req.user as AuthenticatedUser)?.userId;
    if (!userId) throw new ForbiddenException('Missing user identity');
    return this.membersService.getInvites(userId, restaurantId);
  }

  /**
   * MEMBER-04: Update a member's role (owner only).
   */
  @Patch(':restaurantId/members/:memberId')
  async updateMemberRole(
    @Req() req: Request & { user: AuthenticatedUser },
    @Param('restaurantId') restaurantId: string,
    @Param('memberId') memberId: string,
    @Body() body: UpdateMemberRoleDto,
  ): Promise<void> {
    const userId = (req.user as AuthenticatedUser)?.userId;
    if (!userId) throw new ForbiddenException('Missing user identity');
    return this.membersService.updateMemberRole(userId, restaurantId, memberId, body.role);
  }

  /**
   * MEMBER-05: Hard-delete a member's URA row (owner only).
   * User account and all historical data preserved.
   * Per CONTEXT D-15: hard delete, not soft deactivation.
   */
  @Delete(':restaurantId/members/:memberId')
  async removeMember(
    @Req() req: Request & { user: AuthenticatedUser },
    @Param('restaurantId') restaurantId: string,
    @Param('memberId') memberId: string,
  ): Promise<void> {
    const userId = (req.user as AuthenticatedUser)?.userId;
    if (!userId) throw new ForbiddenException('Missing user identity');
    return this.membersService.removeMember(userId, restaurantId, memberId);
  }

  /**
   * MEMBER-10: Manager-initiated add of existing user (by email).
   * Enables adding OAuth users who have an account but no password.
   * Body: { email: string, role: 'owner' | 'manager' | 'staff' }
   */
  @Post(':restaurantId/members')
  async addMember(
    @Req() req: Request & { user: AuthenticatedUser },
    @Param('restaurantId') restaurantId: string,
    @Body() body: { email: string; role: 'owner' | 'manager' | 'staff' },
  ): Promise<void> {
    const userId = (req.user as AuthenticatedUser)?.userId;
    if (!userId) throw new ForbiddenException('Missing user identity');
    return this.membersService.addMember(userId, restaurantId, body.email, body.role);
  }

  /**
   * Revoke a pending invite (owner or manager).
   * DELETE /restaurants/:restaurantId/invites/:code
   */
  @Delete(':restaurantId/invites/:code')
  async revokeInvite(
    @Req() req: Request & { user: AuthenticatedUser },
    @Param('restaurantId') restaurantId: string,
    @Param('code') code: string,
  ): Promise<void> {
    const userId = (req.user as AuthenticatedUser)?.userId;
    if (!userId) throw new ForbiddenException('Missing user identity');
    return this.membersService.revokeInvite(userId, restaurantId, code);
  }
}

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| JWT → controller | User identity and role from JWT; userId from req.user |
| controller → service | actorUserId validated by JwtAuthGuard before reaching service |
| service → Supabase | service-role key bypasses RLS; fine-grained auth via assertMembership() |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-33-03-01 | Spoofing | Non-member accessing /restaurants/:id/members | mitigate | assertMembership() throws ForbiddenException if no active URA row for actorUserId + restaurantId |
| T-33-03-02 | Elevation of Privilege | Staff calling PATCH /members/:userId to change roles | mitigate | assertMembership called with 'owner' required; ForbiddenException thrown for non-owners |
| T-33-03-03 | Denial of Service | Removing last owner locks restaurant | mitigate | Last-owner count check in removeMember and updateMemberRole; BadRequestException with clear message |
| T-33-03-04 | Spoofing | Manager adding another user as 'owner' via POST /members | mitigate | addMember checks actorAccess.role === 'manager' → only 'staff' allowed; ForbiddenException otherwise |
| T-33-03-05 | Information Disclosure | Pending invite codes visible to staff | mitigate | getInvites calls assertMembership with 'owner|manager'; staff get ForbiddenException |
