```python
CLASS_NAMES = {
    0:"wine_entry", 1:"section_header", 2:"wine_name", 3:"producer",
    4:"vintage", 5:"price", 6:"grape_variety", 7:"origin_info",
    8:"description", 9:"serving_type", 10:"rating", 11:"classification",
    12:"bottle_info"
}

def generate_stats(labels_base: Path) -> dict:
    from collections import defaultdict
    counts = defaultdict(lambda: {"train":0, "val":0, "test":0})
    for split in ("train", "val", "test"):
        for lbl_file in (labels_base / split).glob("*.txt"):
            for line in lbl_file.read_text().strip().splitlines():
                if line:
                    cid = int(line.split()[0])
                    counts[CLASS_NAMES[cid]][split] += 1
    return {"class_distribution": {k: dict(v) for k, v in counts.items()}}
```

DATA-05 augmentation hyperparameters (ultralytics built-in — no disk pre-augmentation needed):
  fliplr: 0.5     # horizontal flip
  degrees: 10     # rotation range ±10°
  hsv_v: 0.4      # brightness variation
  mosaic: 1.0     # mosaic augmentation (4-image combine)
