    assert isinstance(result, dict)

def test_run_layer1_inference_skips_high_confidence_fields():
    """Fields with confidence >= 0.8 are NOT overwritten by Layer 1 (D-08)."""
    fc = {"country": {"value": "France", "confidence": 0.95, "source": "visible"}}
    result = run_layer1_inference(fc, "Château Margaux 2015")
    # country should NOT be in result since it's already high confidence
    if "country" in result:
        assert result["country"]["confidence"] > 0.95  # Only if strictly better

def test_run_layer1_inference_infers_when_field_absent():
    """Field absent from fc → eligible for deterministic inference."""
    assert _should_infer({}, "country") is True
    assert _should_infer({"country": {"value": "X", "confidence": 0.3}}, "country") is True
    assert _should_infer({"country": {"value": "X", "confidence": 0.95}}, "country") is False
```

# ===========================================================================
# BoundedLRUCache (D-04, D-06 bug #10)
# ===========================================================================

def test_bounded_lru_cache_eviction():
    """Cache with max_size=3 evicts oldest entry when 4th is added."""
    cache = BoundedLRUCache(max_size=3)
    cache.put("a", 1)
    cache.put("b", 2)
    cache.put("c", 3)
    assert cache.get("a") == 1
    cache.put("d", 4)  # Should evict "b" (oldest not recently accessed)
    assert len(cache) == 3
    assert cache.get("d") == 4

def test_bounded_lru_cache_ttl_expiry():
    """Expired entry returns None."""
    cache = BoundedLRUCache(max_size=10)
    cache.put("key", "value")
    # With a very small TTL, it should expire
    result = cache.get("key", ttl_seconds=0)  # 0 seconds TTL → expired
    assert result is None
```

def test_merge_conflict_candidates_deep_merge():

def test_merge_conflict_candidates_deduplicates():
    """Same source_url for same field not duplicated."""

    new = {"region": [{"value": "A", "source_url": "https://a.com"}, {"value": "B", "source_url": "https://b.com"}]}
