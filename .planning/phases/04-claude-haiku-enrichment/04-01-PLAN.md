**HaikuEnrichmentService class:**
```python
class HaikuEnrichmentService:
    MODEL = "claude-haiku-4-5-20251001"
    MAX_TOKENS = 512

    def __init__(self):
        self._settings = get_settings()
        self._logger = logging.getLogger(__name__)

    def _get_supabase(self):
        return create_client(self._settings.supabase_url, self._settings.supabase_key)

    def _get_anthropic(self):
        return anthropic.AsyncAnthropic(api_key=self._settings.claude_api_key)

    async def _is_already_enriched(self, wine_id: str, wine_name: str) -> bool:
        """D-05: Check both tables. Return True if enrichment should be skipped."""
        supabase = self._get_supabase()

        # Check 1: submissions row — did Vision already return all 3 fields?
        sub_resp = (
            supabase.table("master_wine_library_submissions")
            .select("payload")
            .eq("id", wine_id)
            .maybe_single()
            .execute()
        )
        if sub_resp.data:
            payload = sub_resp.data.get("payload") or {}
            if all([payload.get("region"), payload.get("country"), payload.get("grape_variety")]):
                self._logger.info(f"Skipping enrichment for {wine_id}: submission already complete")
                return True

        # Check 2: master library — does approved record exist with full data?
        lib_resp = (
            supabase.table("master_wine_library")
            .select("id, region, country, grape_variety")
            .ilike("name", wine_name)
            .limit(1)
            .execute()
        )
        if lib_resp.data:
            rec = lib_resp.data[0]
            if all([rec.get("region"), rec.get("country"), rec.get("grape_variety")]):
                self._logger.info(f"Skipping enrichment for {wine_id}: master library record complete")
                return True

        return False

    async def enrich(
        self,
        wine_id: str,
        wine_name: str,
        vintage: Optional[str],
    ) -> Optional[EnrichmentResult]:
        """
        Main entry point. Returns EnrichmentResult or None if skipped.
        Raises ValueError on malformed Haiku response (let Celery retry).
        """
        if await self._is_already_enriched(wine_id, wine_name):
            return None

        vintage_str = f", vintage {vintage}" if vintage else ""
        prompt = (
            f"You are a wine expert. Given the wine name and vintage below, return ONLY a JSON object "
            f"with exactly these keys: region, country, grape_variety, producer_bio. "
            f"If a field is unknown, use null. Do NOT include any other text or markdown.\n\n"
            f"Wine: {wine_name}{vintage_str}"
        )

        client = self._get_anthropic()
        response = await client.messages.create(
            model=self.MODEL,
            max_tokens=self.MAX_TOKENS,
            messages=[{"role": "user", "content": prompt}],
        )

        raw = response.content[0].text.strip()
        # Strip markdown code fences if present
        if raw.startswith("```"):
            raw = raw.split("```")[1]
            if raw.startswith("json"):
                raw = raw[4:]

        try:
            data = json.loads(raw)
        except json.JSONDecodeError as e:
            raise ValueError(f"Haiku returned non-JSON for wine {wine_id}: {raw!r}") from e

        return EnrichmentResult(
            wine_id=wine_id,
            region=data.get("region"),
            country=data.get("country"),
            grape_variety=data.get("grape_variety"),
            producer_bio=data.get("producer_bio"),
        )
```
