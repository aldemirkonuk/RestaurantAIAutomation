```python
    try:
        await _orchestrator.initialize()
        await _orchestrator.start_all_agents()
        logger.info("All CORE agents started successfully.")
    except Exception as exc:
        logger.error("Agent startup failed: %s", exc)

    yield  # App serves requests here

    logger.info("Shutting down agents…")
    if _orchestrator:
        await _orchestrator.stop_all_agents()
    await bus.disconnect()
    logger.info("Agent shutdown complete.")
```

```python
"""
POS Integration Routes
======================
Exposes the Toast webhook endpoint so incoming POS events can be routed
to the running POSIntegrationAgent instance.

Route: POST /api/v1/pos/webhook/toast
"""

import logging
from typing import Optional

from fastapi import APIRouter, Header, HTTPException, Request

logger = logging.getLogger(__name__)

router = APIRouter(prefix="/api/v1/pos", tags=["POS Integration"])


@router.post("/webhook/toast", status_code=200)
async def toast_webhook(
    request: Request,
    toast_signature: Optional[str] = Header(None, alias="Toast-Signature"),
):
    """
    Receive Toast POS webhook events.

    Toast sends an HMAC-SHA256 signature in the Toast-Signature header.
    The raw request body is passed to POSIntegrationAgent for verification
    so the signature check uses the original bytes (not a re-serialized dict).

    Returns:
        200 {"status": "accepted"}  — event queued for processing
        401 {"detail": "..."}       — HMAC verification failed
        503 {"detail": "..."}       — POSIntegrationAgent not running
    """
    # Import here to avoid circular import (main imports pos_routes,
    # pos_routes imports get_orchestrator from main).
    from main import get_orchestrator  # noqa: PLC0415

    raw_payload: bytes = await request.body()

    # Parse JSON — FastAPI has already read the body so we parse from bytes.
    import json as _json
    try:
        webhook_data = _json.loads(raw_payload)
    except _json.JSONDecodeError as exc:
        logger.warning("Malformed JSON in Toast webhook: %s", exc)
        raise HTTPException(status_code=422, detail=f"Invalid JSON: {exc}")

    orchestrator = get_orchestrator()
    if orchestrator is None:
        logger.error("Webhook received but orchestrator is not running.")
        raise HTTPException(
            status_code=503, detail="Agent orchestrator not running."
        )

    agent = orchestrator.agents.get("pos_integration_agent")
    if agent is None:
        logger.error("pos_integration_agent not found in running agents.")
        raise HTTPException(
            status_code=503, detail="POS integration agent not running."
        )

    try:
        result = await agent.process_toast_webhook(
            webhook_data=webhook_data,
            signature=toast_signature,
            raw_payload=raw_payload,
        )
    except Exception as exc:
        logger.exception("Unexpected error processing Toast webhook: %s", exc)
        raise HTTPException(status_code=500, detail=str(exc))

    # If the agent returned an error status, surface it as HTTP 4xx/5xx
    if isinstance(result, dict) and result.get("status") == "error":
        reason = result.get("reason", "unknown error")
        # Signature failures → 401, other agent errors → 422
        if "signature" in reason.lower() or "hmac" in reason.lower():
            raise HTTPException(status_code=401, detail=reason)
        raise HTTPException(status_code=422, detail=reason)

    return result
```

This file must NOT import from `main` at module level (only inside the function body) to avoid the circular import. The `from main import get_orchestrator` inside the function body is resolved at call time, not import time, so the circular dependency is safe.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Internet → POST /api/v1/pos/webhook/toast | Unauthenticated HTTP requests from Toast (or anyone) cross into agent pipeline |
| webhook route → POSIntegrationAgent | Raw bytes and parsed JSON handed to agent; agent performs HMAC check |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-21-02-01 | Spoofing | Toast-Signature header | mitigate | Raw bytes passed to `agent.process_toast_webhook()` which already implements HMAC-SHA256 verification (built in Phase 19/20). Invalid signature → agent returns `{"status": "error", "reason": "signature..."}` → route raises HTTP 401. |
| T-21-02-02 | Denial of Service | POST /api/v1/pos/webhook/toast — no rate limit | accept | MVP scope. Toast sends at most ~100 webhooks/hour for a single restaurant. FastAPI's default concurrency handles this. Add rate limiting in Phase 22. |
| T-21-02-03 | Tampering | RabbitMQ connection in lifespan | mitigate | Connection failure degrades gracefully (warning log, agents don't start, HTTP routes still serve). App does not crash on RabbitMQ unavailability. |
| T-21-02-04 | Information Disclosure | Exception details surfaced in HTTP 500 response | mitigate | `str(exc)` in HTTP 500 detail. Acceptable for MVP where this is an internal service. In Phase 22, replace with generic message + Sentry event. |
</threat_model>
