```typescript
// ==========================================================================
// iCAL SUBSCRIPTION FEED (D-07, D-08, D-09)
// ==========================================================================

async getOrGenerateICalToken(restaurantId: string): Promise<string> {
  const { data, error } = await this.databaseService.supabase
    .from('restaurants')
    .select('calendar_ical_token')
    .eq('id', restaurantId)
    .single();

  if (error) {
    throw new Error(`Failed to fetch iCal token: ${error.message}`);
  }

  if (data?.calendar_ical_token) {
    return data.calendar_ical_token;
  }

  const token = crypto.randomBytes(32).toString('hex');
  const { error: updateError } = await this.databaseService.supabase
    .from('restaurants')
    .update({ calendar_ical_token: token })
    .eq('id', restaurantId);

  if (updateError) {
    throw new Error(`Failed to store iCal token: ${updateError.message}`);
  }

  return token;
}

async regenerateICalToken(restaurantId: string): Promise<string> {
  const token = crypto.randomBytes(32).toString('hex');
  const { error } = await this.databaseService.supabase
    .from('restaurants')
    .update({ calendar_ical_token: token })
    .eq('id', restaurantId);

  if (error) {
    throw new Error(`Failed to regenerate iCal token: ${error.message}`);
  }

  return token;
}

async getICalFeed(token: string): Promise<string> {
  // Look up restaurant by token
  const { data: restaurant, error: restError } = await this.databaseService.supabase
    .from('restaurants')
    .select('id, name')
    .eq('calendar_ical_token', token)
    .single();

  if (restError || !restaurant) {
    // Return empty calendar (not 404) to avoid exposing token validity
    const emptyCal = ical({ name: 'WineOps Calendar', prodId: '-//WineOps//Restaurant Calendar//EN' });
    return emptyCal.toString();
  }

  // Fetch all events for this restaurant
  const { data: events, error: eventsError } = await this.databaseService.supabase
    .from('calendar_events')
    .select('id, title, description, start_date, start_time, end_date, end_time, all_day, status')
    .eq('restaurant_id', restaurant.id)
    .order('start_date', { ascending: true });

  if (eventsError || !events) {
    const emptyCal = ical({ name: 'WineOps Calendar', prodId: '-//WineOps//Restaurant Calendar//EN' });
    return emptyCal.toString();
  }

  const calendar = ical({
    name: `${restaurant.name || 'WineOps'} Calendar`,
    prodId: '-//WineOps//Restaurant Calendar//EN',
  });

  for (const event of events) {
    // Map status to RFC 5545 VEVENT STATUS
    const icalStatus = (event.status === 'cancelled' || event.status === 'dismissed')
      ? 'CANCELLED'
      : (event.status === 'pending')
      ? 'TENTATIVE'
      : 'CONFIRMED';

    const startDateStr = event.start_date;
    const endDateStr = event.end_date || event.start_date;

    let startDate: Date;
    let endDate: Date;

    if (event.all_day) {
      startDate = new Date(`${startDateStr}T00:00:00`);
      endDate = new Date(`${endDateStr}T00:00:00`);
      // For all-day events, DTEND is exclusive (next day per RFC 5545)
      endDate.setDate(endDate.getDate() + 1);
    } else {
      const startTime = event.start_time || '00:00';
      const endTime = event.end_time || '23:59';
      startDate = new Date(`${startDateStr}T${startTime}:00`);
      endDate = new Date(`${endDateStr}T${endTime}:00`);
    }

    calendar.createEvent({
      id: `${event.id}@wineops.app`,
      start: startDate,
      end: endDate,
      summary: event.title,
      description: event.description || undefined,
      allDay: event.all_day,
      status: icalStatus as any,
    });
  }

  return calendar.toString();
}
```

```typescript
// ==========================================================================
// iCAL SUBSCRIPTION FEED (D-07, D-08, D-09)

@Get('feed/:token.ics')
@Public()
@ApiOperation({ summary: 'Public iCal feed — subscribe with Outlook/Apple/Google Calendar' })
@ApiParam({ name: 'token', description: 'Restaurant iCal token (64-char hex)' })
async getICalFeed(
  @Param('token') token: string,
  @Res() res: Response,
): Promise<void> {
  const icalString = await this.calendarService.getICalFeed(token);
  res.setHeader('Content-Type', 'text/calendar; charset=utf-8');
  res.setHeader('Content-Disposition', 'attachment; filename="wineops-calendar.ics"');
  res.setHeader('Cache-Control', 'no-cache, no-store');
  res.send(icalString);
}

@Get('ical-token')
@ApiOperation({ summary: 'Get or generate iCal subscription token for current restaurant' })
@ApiResponse({ status: 200, type: ICalTokenResponseDto })
async getICalToken(
  @CurrentUser() user: { userId: string; restaurantId: string },
): Promise<ICalTokenResponseDto> {
  const token = await this.calendarService.getOrGenerateICalToken(user.restaurantId);
  const feedUrl = `/api/v1/calendar/feed/${token}.ics`;
  return { token, feedUrl };
}

@Post('ical-token/regenerate')
@ApiOperation({ summary: 'Regenerate iCal token — invalidates all existing subscriptions' })
@ApiResponse({ status: 201, type: ICalTokenResponseDto })
async regenerateICalToken(
  @CurrentUser() user: { userId: string; restaurantId: string },
): Promise<ICalTokenResponseDto> {
  const token = await this.calendarService.regenerateICalToken(user.restaurantId);
  const feedUrl = `/api/v1/calendar/feed/${token}.ics`;
  return { token, feedUrl };
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Public internet → GET /calendar/feed/:token.ics | No auth header; token in URL path is the only authentication |
| Authenticated user → GET /calendar/ical-token | JWT-authenticated, generates/returns token for their restaurant only |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-30-07 | Information Disclosure | Token in server logs | mitigate | `res.setHeader('Cache-Control', 'no-cache')` prevents proxy caching; NestJS default logger logs path — document in SUMMARY that `/calendar/feed/` routes should be excluded from log aggregation (see Pitfall 3 in RESEARCH.md). No code change needed in Phase 30 |
| T-30-08 | Spoofing | Token enumeration brute force | accept | 256-bit entropy (crypto.randomBytes(32)) = 2^256 space; brute force infeasible. Standard rate limiting (if any) on api-gateway applies |
| T-30-09 | Information Disclosure | Empty calendar on invalid token | mitigate | Return empty calendar (not 404/401) so attackers cannot distinguish valid vs invalid tokens via status codes |
| T-30-10 | Elevation of Privilege | Accessing another restaurant's feed | mitigate | Token is restaurant-scoped; only events for the token's restaurant are returned; cross-restaurant access impossible without the token |
</threat_model>
