```python
def _current_production(img):
    from PIL import Image, ImageEnhance
    if img.mode != "RGB": img = img.convert("RGB")
    w, h = img.size
    if max(w, h) < 1200:
        scale = 1200 / max(w, h)
        img = img.resize((int(w * scale), int(h * scale)), Image.LANCZOS)
    return ImageEnhance.Contrast(img).enhance(1.3)

def _upscale_2400(img):
    # Same as current_production but threshold 2400px instead of 1200px
    from PIL import Image, ImageEnhance
    if img.mode != "RGB": img = img.convert("RGB")
    w, h = img.size
    if max(w, h) < 2400:
        scale = 2400 / max(w, h)
        img = img.resize((int(w * scale), int(h * scale)), Image.LANCZOS)
    return ImageEnhance.Contrast(img).enhance(1.3)

def _contrast(img, factor):
    from PIL import Image, ImageEnhance
    if img.mode != "RGB": img = img.convert("RGB")
    w, h = img.size
    if max(w, h) < 1200:
        scale = 1200 / max(w, h)
        img = img.resize((int(w * scale), int(h * scale)), Image.LANCZOS)
    return ImageEnhance.Contrast(img).enhance(factor)

def _sharpen(img):
    from PIL import Image, ImageEnhance, ImageFilter
    if img.mode != "RGB": img = img.convert("RGB")
    w, h = img.size
    if max(w, h) < 1200:
        scale = 1200 / max(w, h)
        img = img.resize((int(w * scale), int(h * scale)), Image.LANCZOS)
    img = ImageEnhance.Contrast(img).enhance(1.3)
    return img.filter(ImageFilter.SHARPEN)

def _no_upscale(img):
    from PIL import Image, ImageEnhance
    if img.mode != "RGB": img = img.convert("RGB")
    return ImageEnhance.Contrast(img).enhance(1.3)

def _grayscale(img):
    from PIL import Image, ImageEnhance
    if img.mode != "RGB": img = img.convert("RGB")
    w, h = img.size
    if max(w, h) < 1200:
        scale = 1200 / max(w, h)
        img = img.resize((int(w * scale), int(h * scale)), Image.LANCZOS)
    img = img.convert("L").convert("RGB")
    return ImageEnhance.Contrast(img).enhance(1.3)
```
