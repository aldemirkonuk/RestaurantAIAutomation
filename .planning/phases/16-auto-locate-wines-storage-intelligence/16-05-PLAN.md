  const result = computeAutoLocatePlan(
    mergedInventory,      // ALL wines (D3 — not filtered)

  setAutoLocateResult(result)
  setShowAutoLocateModal(true)

// handleConfirmAutoLocate: called when modal confirms
function handleConfirmAutoLocate(selected: WineLocationScore[]) {
  for (const assignment of selected) {
    assignWineToLocation(assignment.wineId, assignment.locationId, assignment.quantity)
  }
  // Invalidate after all batch calls
  queryClient.invalidateQueries({ queryKey: ['winesAtLocation', restaurantId] })
  queryClient.invalidateQueries({ queryKey: ['storageLocationMappings', restaurantId] })
  setShowAutoLocateModal(false)
  setAutoLocateResult(null)
}
