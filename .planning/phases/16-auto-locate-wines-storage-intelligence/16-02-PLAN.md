    import { useState, useRef, useEffect } from 'react'
    import { MapPin, ChevronDown, Loader2, X } from 'lucide-react'
    import type { StorageLocation } from '../../hooks/useStorageLocations'

    interface LocationPickerCellProps { ... }

    export function LocationPickerCell({ wineId, quantity, locations, currentLocation, onAssign, onRemove }: LocationPickerCellProps) {
      const [isOpen, setIsOpen] = useState(false)
      const [isLoading, setIsLoading] = useState(false)
      const ref = useRef<HTMLDivElement>(null)

      useEffect(() => {
        const handler = (e: MouseEvent) => {
          if (ref.current && !ref.current.contains(e.target as Node)) setIsOpen(false)

        </div>
      )
    }
