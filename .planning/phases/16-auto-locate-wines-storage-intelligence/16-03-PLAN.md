<!-- How to call it in Inventory.tsx row render -->
```tsx
// item.id = master_wine_library UUID (correct for assignWineToLocation)
// item.liveStock + item.shadowStock = total stock for quantity
// getWineLocation(item.id) = current StorageLocation | null
// storageLocations = StorageLocation[] from useStorageLocations()
<LocationPickerCell
  wineId={item.id}
  quantity={(item.liveStock || 0) + (item.shadowStock || 0) || 1}
  locations={storageLocations}
  currentLocation={getWineLocation(item.id)}
  onAssign={(locationId, qty) => {
    assignWineToLocation(item.id, locationId, qty)
    setInventory(prev => prev.map(inv =>
      inv.id === item.id
        ? { ...inv, storageLocation: storageLocations.find(l => l.id === locationId)?.name }
        : inv
    ))
  }}
  onRemove={() => {
    removeWineFromLocation(item.id)
    setInventory(prev => prev.map(inv =>
      inv.id === item.id ? { ...inv, storageLocation: undefined } : inv
    ))
  }}
/>
```
