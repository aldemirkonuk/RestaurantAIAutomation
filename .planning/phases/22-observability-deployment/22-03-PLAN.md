"""
Health & Metrics Routes
========================
Admin-key-protected endpoints for monitoring agent health and system metrics.

Auth: X-Admin-Key header matching ADMIN_API_KEY env var (server-to-server only).
      Pattern from research_routes.py:34-41.

IMPORTANT: GET /health (public, no auth) lives in main.py line 92.
           Do NOT define it here — FastAPI will raise AssertionError on duplicate routes.
"""

import logging
import os
from typing import Optional

from fastapi import APIRouter, Depends, Header, HTTPException

from config.settings import get_settings

logger = logging.getLogger(__name__)

router = APIRouter(tags=["Health & Metrics"])


def verify_admin_key(x_admin_key: Optional[str] = Header(None)) -> str:
    """Require X-Admin-Key header matching ADMIN_API_KEY env var."""
    expected = os.getenv("ADMIN_API_KEY", "")
    if not x_admin_key:
        raise HTTPException(status_code=401, detail="Invalid or missing X-Admin-Key")
    if not expected or x_admin_key != expected:
        raise HTTPException(status_code=401, detail="Invalid or missing X-Admin-Key")
    return x_admin_key


@router.get("/api/v1/health/agents", dependencies=[Depends(verify_admin_key)])
async def get_all_agents_health():
    """Return health status for all running agents."""
    from main import get_orchestrator  # noqa: PLC0415

    orchestrator = get_orchestrator()
    if orchestrator is None:
        return {"agents": [], "orchestrator_status": "not_running", "count": 0}

    agents = []
    for name, agent in orchestrator.agents.items():
        try:
            health = agent.get_health()
            agents.append(health)
        except Exception as exc:
            logger.warning("get_health() failed for agent %s: %s", name, exc)
            agents.append({"agent_name": name, "status": "error", "healthy": False})

    return {"agents": agents, "count": len(agents)}


@router.get("/api/v1/health/agents/{name}", dependencies=[Depends(verify_admin_key)])
async def get_agent_health(name: str):
    """Return detailed metrics for a specific agent by name."""
    from main import get_orchestrator  # noqa: PLC0415

    orchestrator = get_orchestrator()
    if orchestrator is None:
        raise HTTPException(status_code=503, detail="Orchestrator not running")

    agent = orchestrator.agents.get(name)
    if agent is None:
        raise HTTPException(
            status_code=404, detail=f"Agent '{name}' not found. "
            f"Running agents: {list(orchestrator.agents.keys())}"
        )

    return agent.get_detailed_health()


@router.get("/api/v1/metrics", dependencies=[Depends(verify_admin_key)])
async def get_system_metrics():
    """
    Return system metrics: DLQ size, active sagas, per-agent message counts.

    Combines AgentOrchestrator.get_metrics() (already implemented) with
    two Supabase count queries for DLQ and active saga counts.
    """
    from main import get_orchestrator  # noqa: PLC0415

    orchestrator = get_orchestrator()
    if orchestrator is None:
        raise HTTPException(status_code=503, detail="Orchestrator not running")

    metrics = orchestrator.get_metrics()

    # Augment with DLQ and saga counts — not included in existing get_metrics()
    settings = get_settings()
    dlq_size = 0
    active_sagas = 0
    try:
        if settings.supabase_client:
            dlq_result = (
                settings.supabase_client
                .table("dead_letter_queue")
                .select("id", count="exact")
                .execute()
            )
            dlq_size = dlq_result.count or 0

            saga_result = (
                settings.supabase_client
                .table("saga_state")
                .select("id", count="exact")
                .eq("status", "running")
                .execute()
            )
            active_sagas = saga_result.count or 0
    except Exception as exc:
        logger.warning("Failed to query DLQ/saga counts from Supabase: %s", exc)

    metrics["dlq_size"] = dlq_size
    metrics["active_sagas"] = active_sagas

    return metrics
```

"""
POS Integration Routes
======================
Generic webhook endpoint dispatching to registered POS provider adapters.

Route: POST /api/v1/pos/webhook/{provider}
Currently supported: toast (via ToastAdapter)

To add a new provider: create adapters/<name>_adapter.py implementing
POSProvider, then add an entry to _PROVIDERS below. No agent changes needed.
"""

import json as _json
import logging
from typing import Optional

from fastapi import APIRouter, Header, HTTPException, Request

from adapters.toast_adapter import ToastAdapter
from core.pos_provider import POSProvider

logger = logging.getLogger(__name__)

router = APIRouter(prefix="/api/v1/pos", tags=["POS Integration"])

# Provider registry — add new adapters here as they are implemented
_PROVIDERS: dict[str, POSProvider] = {
    "toast": ToastAdapter(),
}


@router.post("/webhook/{provider}", status_code=200)
async def pos_webhook(
    provider: str,
    request: Request,
    toast_signature: Optional[str] = Header(None, alias="Toast-Signature"),
):
    """
    Receive POS webhook events from any registered provider.

    Returns:
        200 {"status": "accepted"}  — event queued for processing
        401 {"detail": "..."}       — signature verification failed
        404 {"detail": "..."}       — unknown provider
        503 {"detail": "..."}       — agent not running
    """
    if provider not in _PROVIDERS:
        raise HTTPException(
            status_code=404,
            detail=f"Unknown POS provider: '{provider}'. "
            f"Supported: {list(_PROVIDERS.keys())}",
        )

    adapter = _PROVIDERS[provider]
    raw_payload: bytes = await request.body()

    # Verify webhook signature via adapter (HMAC check)
    if not await adapter.verify_webhook(raw_payload, toast_signature or ""):
        logger.warning(
            "Webhook signature verification failed for provider: %s", provider
        )
        raise HTTPException(
            status_code=401, detail="Webhook signature verification failed"
        )

    # Parse JSON payload
    try:
        webhook_data = _json.loads(raw_payload)
    except _json.JSONDecodeError as exc:
        logger.warning("Malformed JSON in %s webhook: %s", provider, exc)
        raise HTTPException(status_code=422, detail=f"Invalid JSON: {exc}")

    # Route to POSIntegrationAgent (still receives raw data for backward compat
    # with process_toast_webhook — full POSEvent migration is Wave 6 scope)
    from main import get_orchestrator  # noqa: PLC0415

    orchestrator = get_orchestrator()
    if orchestrator is None:
        logger.error("Webhook received but orchestrator is not running.")
        raise HTTPException(
            status_code=503, detail="Agent orchestrator not running."
        )

    agent = orchestrator.agents.get("pos_integration_agent")
    if agent is None:
        logger.error("pos_integration_agent not found in running agents.")
        raise HTTPException(
            status_code=503, detail="POS integration agent not running."
        )

    try:
        result = await agent.process_toast_webhook(
            webhook_data=webhook_data,
            signature=toast_signature,
            raw_payload=raw_payload,
        )
    except Exception as exc:
        logger.exception("Unexpected error processing %s webhook: %s", provider, exc)
        raise HTTPException(status_code=500, detail=str(exc))

    if isinstance(result, dict) and result.get("status") == "error":
        reason = result.get("reason") or result.get("message", "unknown error")
        if "signature" in reason.lower() or "hmac" in reason.lower():
            raise HTTPException(status_code=401, detail=reason)
        raise HTTPException(status_code=422, detail=reason)

    return result
```
