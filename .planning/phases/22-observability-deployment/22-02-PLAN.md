```python
"""
POS Provider Abstraction
========================
Protocol and shared data models for all POS system adapters.

To add a new POS system (Square, Clover, etc.):
  1. Create adapters/<name>_adapter.py implementing POSProvider
  2. Add entry to _PROVIDERS dict in api/pos_routes.py
  No changes to agents required.
"""

from datetime import datetime
from typing import Protocol, runtime_checkable

from pydantic import BaseModel


class POSEvent(BaseModel):
    """Normalized POS event — common schema across all POS systems."""

    event_type: str
    restaurant_guid: str
    timestamp: datetime
    items: list[dict]
    raw_payload: dict


@runtime_checkable
class POSProvider(Protocol):
    """Protocol all POS adapters must implement."""

    async def verify_webhook(self, raw: bytes, signature: str) -> bool:
        """Verify the webhook signature. Return True if valid, False if invalid."""
        ...

    async def normalize_event(self, raw: dict) -> POSEvent:
        """Normalize a raw POS-specific event payload into a POSEvent."""
        ...
```

```python
"""
Toast POS Adapter
=================
Implements POSProvider for Toast POS system.

Handles HMAC-SHA256 webhook signature verification and event normalization.
All Toast-specific logic lives here — POSIntegrationAgent only sees POSEvent.
"""

import hashlib
import hmac
import logging
import os
from datetime import datetime, timezone

from core.pos_provider import POSEvent

logger = logging.getLogger(__name__)


class ToastAdapter:
    """POSProvider implementation for Toast POS."""

    async def verify_webhook(self, raw: bytes, signature: str) -> bool:
        """
        Verify Toast webhook HMAC-SHA256 signature.

        Toast sends: Toast-Signature: sha256=<hex_digest>

        Behavior:
        - TOAST_WEBHOOK_SECRET not set → warn + return True (dev mode)
        - signature missing or wrong → return False (401 in route)
        - signature matches → return True
        """
        secret = os.getenv("TOAST_WEBHOOK_SECRET", "")
        if not secret:
            logger.warning(
                "TOAST_WEBHOOK_SECRET not set — skipping HMAC verification (dev mode)"
            )
            return True

        if not signature:
            return False

        # Toast sends 'sha256=<hex>' — strip the prefix
        sig_value = signature.removeprefix("sha256=")

        expected = hmac.new(
            secret.encode("utf-8"), raw, hashlib.sha256
        ).hexdigest()

        return hmac.compare_digest(expected, sig_value)

    async def normalize_event(self, raw: dict) -> POSEvent:
        """
        Normalize a raw Toast webhook payload into a POSEvent.

        Toast webhook top-level fields used:
        - restaurantGuid: identifies the restaurant
        - eventType / type / orderType: event classification
        - orders / checks: list of order/check items
        """
        event_type = (
            raw.get("eventType")
            or raw.get("type")
            or raw.get("orderType", "order.created")
        )
        restaurant_guid = (
            raw.get("restaurantGuid")
            or raw.get("restaurant_guid", "")
        )
        items = raw.get("orders") or raw.get("checks") or []

        return POSEvent(
            event_type=str(event_type),
            restaurant_guid=str(restaurant_guid),
            timestamp=datetime.now(timezone.utc),
            items=items if isinstance(items, list) else [],
            raw_payload=raw,
        )
```
