```python
from unittest.mock import patch

    """Replica of the Sentry init logic from main.py — testable without importing main."""
    import sentry_sdk
    from sentry_sdk.integrations.fastapi import FastApiIntegration
    from sentry_sdk.integrations.starlette import StarletteIntegration

    if not dsn:
        if environment == "production":
            raise ValueError(
                "SENTRY_DSN is required when ENVIRONMENT=production. "
                "Set SENTRY_DSN in Railway dashboard environment variables."
            )
        logging.getLogger(__name__).warning(
            "SENTRY_DSN not set — Sentry disabled. "
            "(Set ENVIRONMENT=production to fail fast on missing DSN)"

        return
```

```python
"""Unit tests for CORS middleware (INFRA-01)."""
import json
import pytest
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from fastapi.testclient import TestClient


def _make_app(allowed_origins):
    app = FastAPI()
    app.add_middleware(
        CORSMiddleware,
        allow_origins=allowed_origins,
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )

    @app.get("/test")
    def test_route():
        return {"ok": True}

    return TestClient(app)


def test_cors_header_present_for_allowed_origin():
    client = _make_app(["http://localhost:5173"])
    resp = client.get("/test", headers={"Origin": "http://localhost:5173"})
    assert resp.status_code == 200
    assert resp.headers.get("access-control-allow-origin") == "http://localhost:5173"


def test_cors_preflight_returns_200():
    client = _make_app(["http://localhost:5173"])
    resp = client.options(
        "/test",
        headers={
            "Origin": "http://localhost:5173",
            "Access-Control-Request-Method": "GET",
        },
    )
    assert resp.status_code == 200


def test_cors_absent_for_unknown_origin():
    client = _make_app(["http://localhost:5173"])
    resp = client.get("/test", headers={"Origin": "http://evil.com"})
    assert "access-control-allow-origin" not in resp.headers


def test_allowed_origins_json_array_parses():
    raw = '["http://localhost:5173", "https://myapp.vercel.app"]'
    origins = json.loads(raw)
    client = _make_app(origins)
    for origin in origins:
        resp = client.get("/test", headers={"Origin": origin})
        assert resp.headers.get("access-control-allow-origin") == origin
```
