## Context

- `apps/api-gateway/src/common/orchestrator/orchestrator.service.ts` already has `httpClient = axios.create({ baseURL: AGENT_ORCHESTRATOR_URL })` — extend it with 3 new proxy methods + `getAdminHeaders()`
- `TenantGuard` is a global guard (app.module.ts line 83). It has a `@TenantBypass()` decorator (`tenant.decorator.ts`) that sets `TENANT_BYPASS_KEY` metadata — TenantGuard checks this and returns `true`. Health proxy routes have no restaurant-scoped resources, so `@TenantBypass()` is required
- JWT auth: use `AuthGuard('jwt')` from `@nestjs/passport` directly — the JWT strategy is registered globally via `AuthModule` in `AppModule`. No need to import `AuthModule` into `OrchestratorModule`
- Two NestJS controllers can live in one file (TypeScript module exports); `orchestrator.module.ts` imports both from `health-proxy.controller.ts`

- `AdminHealth.tsx` copies framer-motion + lucide-react + sonner + axios pattern from `AdminPanel.tsx` (lines 1–47 of AdminPanel)
- `vercel.json` rewrite destination MUST be a literal URL (Vercel does not interpolate env vars in rewrites). Use a clearly marked placeholder — executor must replace with actual Railway api-gateway URL after deploy
- Frontend calls `/api/health/agents` (goes through NestJS proxy at `@Controller('api/health')`), NestJS forwards to Python at `/api/v1/health/agents`

```typescript
import { Controller, Get, Param, UseGuards } from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { TenantBypass } from '../tenant/tenant.decorator';
import { OrchestratorService } from './orchestrator.service';

/**
 * Proxies agent health requests to the Python orchestrator service.
 *
 * Auth: Supabase JWT required (any authenticated user — no role restriction).
 * The X-Admin-Key header is added server-side by OrchestratorService.getAdminHeaders().
 * ADMIN_API_KEY never reaches the browser.
 *
 * @TenantBypass: health routes have no restaurant-scoped resources.
 * Without this, TenantGuard (global guard) would be a no-op anyway for
 * routes with no restaurantId param, but @TenantBypass makes intent explicit.
 */
@TenantBypass()
@UseGuards(AuthGuard('jwt'))
@Controller('api/health')
export class HealthProxyController {
  constructor(private readonly orchestratorService: OrchestratorService) {}

  @Get('agents')
  getAllAgentsHealth(): Promise<any> {
    return this.orchestratorService.getAgentHealthAll();
  }

  @Get('agents/:name')
  getAgentHealth(@Param('name') name: string): Promise<any> {
    return this.orchestratorService.getAgentHealthByName(name);
  }
}

/**
 * Proxies system metrics requests to the Python orchestrator.
 * Same auth + bypass rules as HealthProxyController above.
 */
@TenantBypass()
@UseGuards(AuthGuard('jwt'))
@Controller('api/metrics')
export class MetricsProxyController {
  constructor(private readonly orchestratorService: OrchestratorService) {}

  @Get()
  getSystemMetrics(): Promise<any> {
    return this.orchestratorService.getSystemMetrics();
  }
}
```

### Task 2: Create apps/web/src/pages/AdminHealth.tsx + add route in App.tsx

**File**: `apps/web/src/pages/AdminHealth.tsx`
**Action**: create

**Details** — write this exact file:
```tsx
import { useState, useEffect, useCallback } from 'react'
import { motion } from 'framer-motion'
import {
  Activity,
  Server,
  RefreshCw,
  CheckCircle,
  XCircle,
  AlertCircle,
} from 'lucide-react'
import { toast } from 'sonner'
import axios from 'axios'

const API_GATEWAY_URL = import.meta.env.VITE_API_GATEWAY_URL || 'http://localhost:4000'

const containerVariants = {
  hidden: { opacity: 0 },
  visible: { opacity: 1, transition: { staggerChildren: 0.05 } },
}

const itemVariants = {
  hidden: { opacity: 0, y: 12 },
  visible: { opacity: 1, y: 0, transition: { duration: 0.25 } },
}

interface AgentHealth {
  agent_name: string
  status: string
  healthy: boolean
  version?: string
  capabilities?: string[]
}

interface HealthResponse {
  agents: AgentHealth[]
  count: number
  orchestrator_status?: string
}

export default function AdminHealth() {
  const [agents, setAgents] = useState<AgentHealth[]>([])
  const [orchestratorStatus, setOrchestratorStatus] = useState<string>('unknown')
  const [loading, setLoading] = useState(true)
  const [lastUpdated, setLastUpdated] = useState<Date | null>(null)

  const fetchHealth = useCallback(async () => {
    try {
      const token = localStorage.getItem('accessToken')
      const res = await axios.get<HealthResponse>(
        `${API_GATEWAY_URL}/api/health/agents`,
        { headers: { Authorization: `Bearer ${token}` } },
      )
      const data = res.data
      setAgents(data.agents ?? [])
      setOrchestratorStatus(data.orchestrator_status ?? 'running')
      setLastUpdated(new Date())
    } catch (err: any) {
      const msg =
        err?.response?.status === 401
          ? 'Unauthorized — please log in again'
          : 'Failed to fetch agent health'
      toast.error(msg)
    } finally {
      setLoading(false)
    }
  }, [])

  useEffect(() => {
    fetchHealth()
    const interval = setInterval(fetchHealth, 30_000)
    return () => clearInterval(interval)
  }, [fetchHealth])

  const allHealthy = agents.length > 0 && agents.every((a) => a.healthy)
  const unhealthyCount = agents.filter((a) => !a.healthy).length

  return (
    <div className="min-h-screen">
      {/* Header */}
      <div className="sticky top-0 z-30 bg-white/80 backdrop-blur-md border-b border-gray-200">
        <div className="h-16 px-6 flex items-center justify-between">
          <h1 className="text-xl font-bold text-gray-900 flex items-center gap-2">
            <Activity className="w-5 h-5 text-emerald-500" />
            Agent Health Dashboard
          </h1>
          <div className="flex items-center gap-3">
            {lastUpdated && (
              <span className="text-xs text-gray-400">
                Updated {lastUpdated.toLocaleTimeString()}
              </span>
            )}
            {!loading && (
              <span
                className={`inline-flex items-center gap-1.5 px-2.5 py-1 rounded-full text-xs font-medium ${
                  allHealthy
                    ? 'bg-emerald-100 text-emerald-700'
                    : unhealthyCount > 0
                    ? 'bg-amber-100 text-amber-700'
                    : 'bg-gray-100 text-gray-500'
                }`}
              >
                {allHealthy ? (
                  <CheckCircle className="w-3 h-3" />
                ) : unhealthyCount > 0 ? (
                  <AlertCircle className="w-3 h-3" />
                ) : (
                  <Server className="w-3 h-3" />
                )}
                {allHealthy
                  ? 'All healthy'
                  : unhealthyCount > 0
                  ? `${unhealthyCount} degraded`
                  : 'No agents'}
              </span>
            )}
            <button
              onClick={fetchHealth}
              className="p-2 rounded-lg hover:bg-gray-100 transition-colors"
              title="Refresh now"
            >
              <RefreshCw className={`w-4 h-4 text-gray-500 ${loading ? 'animate-spin' : ''}`} />
            </button>
          </div>
        </div>
      </div>

      {/* Content */}
      <div className="p-6">
        {loading ? (
          <div className="flex items-center justify-center min-h-[60vh]">
            <RefreshCw className="w-6 h-6 text-gray-300 animate-spin" />
          </div>
        ) : orchestratorStatus === 'not_running' ? (
          <div className="flex flex-col items-center justify-center min-h-[60vh] text-center">
            <AlertCircle className="w-12 h-12 text-amber-400 mb-4" />
            <h2 className="text-lg font-semibold text-gray-900 mb-1">
              Orchestrator Not Running
            </h2>
            <p className="text-sm text-gray-500">
              The agent orchestrator is not connected. Check Railway deployment status.
            </p>
          </div>
        ) : agents.length === 0 ? (
          <div className="flex flex-col items-center justify-center min-h-[60vh] text-center">
            <Server className="w-12 h-12 text-gray-300 mb-4" />
            <h2 className="text-lg font-semibold text-gray-900 mb-1">No Agents Running</h2>
            <p className="text-sm text-gray-500">No agents have been registered yet.</p>
          </div>
        ) : (
          <motion.div
            variants={containerVariants}
            initial="hidden"
            animate="visible"
            className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4"
          >
            {agents.map((agent) => (
              <motion.div
                key={agent.agent_name}
                variants={itemVariants}
                className="bg-white rounded-xl border border-gray-200 shadow-sm p-5 hover:shadow-md transition-shadow"
              >
                <div className="flex items-start justify-between mb-3">
                  <div className="flex items-center gap-2 min-w-0">
                    <Server className="w-4 h-4 text-gray-400 flex-shrink-0" />
                    <span className="font-medium text-gray-900 text-sm truncate capitalize">
                      {agent.agent_name.replace(/_/g, ' ')}
                    </span>
                  </div>
                  {agent.healthy ? (
                    <CheckCircle className="w-5 h-5 text-emerald-500 flex-shrink-0" />
                  ) : (
                    <XCircle className="w-5 h-5 text-rose-500 flex-shrink-0" />
                  )}
                </div>

                <div className="flex items-center gap-2 flex-wrap">
                  <span
                    className={`inline-flex items-center px-2 py-0.5 rounded-full text-xs font-medium ${
                      agent.status === 'active'
                        ? 'bg-emerald-100 text-emerald-700'
                        : agent.status === 'idle'
                        ? 'bg-blue-100 text-blue-700'
                        : agent.status === 'degraded'
                        ? 'bg-amber-100 text-amber-700'
                        : 'bg-rose-100 text-rose-700'
                    }`}
                  >
                    {agent.status}
                  </span>
                  {agent.version && (
                    <span className="text-xs text-gray-400">v{agent.version}</span>
                  )}
                </div>
              </motion.div>
            ))}
          </motion.div>
        )}
      </div>
    </div>
  )
}
```
