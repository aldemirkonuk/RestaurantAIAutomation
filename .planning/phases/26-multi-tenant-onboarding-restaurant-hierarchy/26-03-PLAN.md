**organizations.service.ts**:
```typescript
import { Injectable, Logger } from '@nestjs/common';
import { DatabaseService } from '../database/database.service';

export interface RestaurantBranch {
  id: string;
  name: string;
  city: string | null;
}

@Injectable()
export class OrganizationsService {
  private readonly logger = new Logger(OrganizationsService.name);

  constructor(private readonly databaseService: DatabaseService) {}

  async getBranchesForUser(userId: string): Promise<RestaurantBranch[]> {
    // Find all organizations the user belongs to via organization_members
    const { data: memberships, error } = await this.databaseService.supabase
      .from('organization_members')
      .select('organization_id')
      .eq('user_id', userId);

    if (error || !memberships || memberships.length === 0) {
      this.logger.debug(`No org memberships found for user ${userId} — returning single restaurant fallback`);
      // Fallback: return user's direct restaurant (for users who registered before org system)
      const { data: user } = await this.databaseService.supabase
        .from('users')
        .select('restaurant_id')
        .eq('user_id', userId)
        .maybeSingle();
      if (!user?.restaurant_id) return [];
      const { data: restaurant } = await this.databaseService.supabase
        .from('restaurants')
        .select('id, name, city')
        .eq('id', user.restaurant_id)
        .maybeSingle();
      return restaurant ? [{ id: restaurant.id, name: restaurant.name, city: restaurant.city ?? null }] : [];
    }

    const orgIds = memberships.map((m) => m.organization_id);

    // Fetch all restaurants belonging to these organizations
    const { data: restaurants, error: restErr } = await this.databaseService.supabase
      .from('restaurants')
      .select('id, name, city')
      .in('organization_id', orgIds);

    if (restErr || !restaurants) {
      this.logger.error(`Failed to fetch branches for user ${userId}: ${restErr?.message}`);
      return [];
    }

    return restaurants.map((r) => ({
      id: r.id,
      name: r.name,
      city: r.city ?? null,
    }));
  }
}
```

**organizations.controller.ts**:
```typescript
import { Controller, Get, Req, UseGuards, Logger } from '@nestjs/common';
import { OrganizationsService, RestaurantBranch } from './organizations.service';
import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
import { Request } from 'express';

@Controller('organizations')
export class OrganizationsController {
  private readonly logger = new Logger(OrganizationsController.name);

  constructor(private readonly organizationsService: OrganizationsService) {}

  @Get('branches')
  @UseGuards(JwtAuthGuard)
  async getBranches(@Req() req: Request & { user: any }): Promise<RestaurantBranch[]> {
    this.logger.debug(`Fetching branches for user ${req.user.userId}`);
    return this.organizationsService.getBranchesForUser(req.user.userId);
  }
}
```
