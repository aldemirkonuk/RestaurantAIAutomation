```python
_GRAPE_CACHE: Optional[Dict[str, str]] = None  # {lower_name_or_alias: canonical_name}

def _ensure_grape_cache(supabase) -> Dict[str, str]:
    """Load all grape_varieties into a dict mapping any alias/name → canonical_name. Lazy-loaded."""
    global _GRAPE_CACHE
    if _GRAPE_CACHE is not None:
        return _GRAPE_CACHE
    resp = supabase.table("grape_varieties").select("name,canonical_name,aliases").execute()
    cache = {}
    for row in (resp.data or []):
        cache[row["name"].lower()] = row["canonical_name"]
        cache[row["canonical_name"].lower()] = row["canonical_name"]
        for alias in (row.get("aliases") or []):
            cache[alias.lower()] = row["canonical_name"]
    _GRAPE_CACHE = cache
    return cache
```

```python
def get_region_ancestors(region_id: str) -> List[Dict[str, Any]]:
    supabase = _get_supabase()
    ancestors = []
    current_id = region_id
    visited = set()
    for _ in range(10):  # max 10 hops

            break
        visited.add(current_id)
        resp = supabase.table("wine_regions").select("id,name,level,parent_id,country_code,canonical_name").eq("id", current_id).maybe_single().execute()
        if not resp.data:
            break
        row = resp.data

        if row.get("parent_id") is None:
            break  # reached root (country level)
        current_id = row["parent_id"]
    return ancestors
```

```python
Runs AFTER Phase 7 (field_confidence) and Phase 8 (web verification).
```

```python
    def _apply_ontology_autofills(
        self,
        fc: Dict[str, Any],
        appellation_value: Optional[str],
        canonical_grape: Optional[str],
    ) -> tuple[Dict[str, Any], int]:

        D-04: Write ontology-derived facts into field_confidence with confidence=1.0.

          existing_conf = fc.get(field, {}).get("confidence", 0.0) if isinstance(fc.get(field), dict) else 0.0

              autofill_count += 1
        
        Returns (updated_fc, autofill_count).
```

```python
    def _route_critical_failures(
        self,
        wine_id: str,
        failures: List[OntologyCheckFailure],
        fc: Dict[str, Any],
    ) -> None:
        """
        D-03: For each CRITICAL failure:
          1. Insert into field_review_queue with source='ontology', status='pending'
          2. Set auto_blocked=True on the submission
        
        Maps check names to field names for field_review_queue:
          "region_country" → "country"
          "grape_appellation" → "grape_variety"  
          "vintage_plausibility" → "vintage"
          "color_grape" → "color"  (WARNING only — not routed)
        """
        critical_failures = [f for f in failures if f.severity == "critical"]
```
