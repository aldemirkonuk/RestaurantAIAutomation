```python
"""
DigestCronJob — Phase 24 D-04, D-05, D-06
==========================================
Runs hourly. For each restaurant where local time is 8am and digest_enabled=true,
reads Redis digest key, calls Haiku for natural-language summary, sends via EmailClient.

APScheduler pattern from demo/weekly_report_scheduler.py.
"""
from __future__ import annotations
import asyncio
import json
import logging
from datetime import date, datetime
from typing import Optional, List
from zoneinfo import ZoneInfo

from apscheduler.schedulers.asyncio import AsyncIOScheduler
from apscheduler.triggers.cron import CronTrigger

from config.settings import Settings
from utils.logger import setup_logger

logger = setup_logger(__name__)

# Redis key written by GmailWatchService on watch renewal
WATCH_EXPIRY_REDIS_KEY = "gmail:watch:expiration"
DIGEST_TTL_HOURS = 36


class DigestCronJob:
    """
    Hourly cron: sends 8am daily digest to restaurants whose local time is 8.
    """

    def __init__(self, supabase_client, redis_client, email_client, haiku_client):
        self.supabase = supabase_client
        self.redis = redis_client
        self.email_client = email_client
        self.haiku_client = haiku_client
        self.scheduler: Optional[AsyncIOScheduler] = None
        self.settings = Settings()

    async def start(self) -> None:
        """Start the hourly scheduler."""
        self.scheduler = AsyncIOScheduler()
        self.scheduler.add_job(
            self._check_and_send_digests,
            CronTrigger(minute=0),
            id="phase24_digest_job",
            replace_existing=True,
        )
        self.scheduler.start()
        logger.info("DigestCronJob started — hourly digest check active")

    async def stop(self) -> None:
        if self.scheduler and self.scheduler.running:
            self.scheduler.shutdown(wait=False)

    async def _check_and_send_digests(self) -> None:
        """Called every hour. Sends digest for any restaurant at local 8am."""
        try:
            restaurants = await self._get_digest_enabled_restaurants()
            for r in restaurants:
                tz_str = r.get("timezone") or "America/Los_Angeles"
                try:
                    tz = ZoneInfo(tz_str)
                except Exception:
                    tz = ZoneInfo("America/Los_Angeles")
                local_hour = datetime.now(tz).hour
                if local_hour == 8:
                    try:
                        await self._send_digest(r["id"], r, tz)
                    except Exception as e:
                        logger.error(f"Digest send failed for restaurant {r['id']}: {e}")
        except Exception as e:
            logger.error(f"DigestCronJob _check_and_send_digests error: {e}")

    async def _get_digest_enabled_restaurants(self) -> List[dict]:
        """Query restaurants with at least one active user having digest_enabled=true."""
        try:
            result = (self.supabase.table("restaurants")
                      .select("id, name, timezone")
                      .execute())
            return result.data or []
        except Exception as e:
            logger.warning(f"Could not fetch restaurants: {e}")
            return []

    async def _send_digest(self, restaurant_id: str, restaurant: dict, tz) -> None:
        """Build and send the digest email for one restaurant."""
        today_str = date.today().isoformat()
        digest_key = f"digest:{restaurant_id}:{today_str}"

        # Read promo items from Redis
        raw_items = await self.redis.lrange(digest_key, 0, -1)
        promo_items = sorted(
            [json.loads(i) for i in raw_items],
            key=lambda x: x.get("urgency_score", 0),
            reverse=True
        )

        # Check notification_preferences toggles
        prefs = await self._get_prefs(restaurant_id)
        promos_enabled = prefs.get("digest_promos_enabled", True)
        stalled_enabled = prefs.get("digest_stalled_threads_enabled", True)
        gaps_enabled = prefs.get("digest_procurement_gaps_enabled", True)

        # Gmail Watch status (D-06)
        watch_status = await self._get_watch_status()

        # Build sections
        sections: List[str] = []

        if promos_enabled and promo_items:
            promo_lines = []
            for item in promo_items[:5]:
                line = f"• {item.get('vendor','')} — {item.get('product','')} "
                if item.get("discount_pct"):
                    line += f"({item['discount_pct']}% off)"
                line += f" [urgency: {item.get('urgency_score', 0)}/10]"
                promo_lines.append(line)
            sections.append("DEALS TODAY:\n" + "\n".join(promo_lines))

        if stalled_enabled:
            stalled = await self._get_stalled_threads(restaurant_id)
            if stalled:
                sections.append("AWAITING REPLY (>24h):\n" + "\n".join(
                    f"• {s['provider_name']}: \"{s['snippet']}\""
                    for s in stalled[:5]
                ))

        if gaps_enabled:
            gaps = await self._get_procurement_gaps(restaurant_id)
            if gaps:
                sections.append("UPCOMING EVENT GAPS:\n" + "\n".join(
                    f"• {g['event_title']} ({g['days_away']}d): {g['wine_gap']}"
                    for g in gaps[:3]
                ))

        if not sections:
            logger.info(f"Digest empty for restaurant {restaurant_id} — skipping send")
            return

        # Haiku generates natural-language paragraph
        prose = await self._generate_prose(sections, restaurant)

        # Build email
        subject = f"☀️ Your WineOps Morning Digest — {today_str}"
        body = f"""<p style="color:#666;font-size:12px">{watch_status}</p>
<p>{prose}</p>
<hr style="border:1px solid #eee;margin:16px 0">
<pre style="font-family:monospace;font-size:13px;color:#333">{"chr(10)".join(sections)}</pre>
<p style="color:#999;font-size:11px">WineOps · Manage digest preferences in Settings</p>"""

        manager_email = self.settings.manager_email
        if manager_email:
            # Use EmailClient.send_email() or .send() — adapt to actual method name
            try:
                await self.email_client.send_email(
                    to=manager_email,
                    subject=subject,
                    html_body=body,
                )
            except AttributeError:
                # Fallback if method is named differently
                await self.email_client.send(
                    to_email=manager_email,
                    subject=subject,
                    html_body=body,
                )
            logger.info(f"Digest sent to {manager_email} for restaurant {restaurant_id}")

    async def _generate_prose(self, sections: List[str], restaurant: dict) -> str:
        """Call Haiku to generate a natural-language summary paragraph."""
        try:
            content = "\n\n".join(sections)
            response = await self.haiku_client.messages.create(
                model=self.settings.haiku_model,
                max_tokens=512,
                messages=[{
                    "role": "user",
                    "content": (
                        f"Write a brief (2-3 sentence) morning summary for "
                        f"{restaurant.get('name','the restaurant')} based on:\n\n{content}\n\n"
                        "Be specific, mention the most actionable item first. "
                        "Reference urgency scores and calendar proximity where relevant. "
                        "Write as if you're a knowledgeable beverage director assistant."
                    )
                }]
            )
            return response.content[0].text.strip() if response.content else content[:500]
        except Exception as e:
            logger.warning(f"Haiku prose generation failed: {e}")
            return "\n".join(sections[:2])

    async def _get_prefs(self, restaurant_id: str) -> dict:
        try:
            result = (self.supabase.table("notification_preferences")
                      .select("digest_promos_enabled,digest_stalled_threads_enabled,digest_procurement_gaps_enabled")
                      .eq("restaurant_id", restaurant_id)
                      .limit(1).execute())
            return result.data[0] if result.data else {}
        except Exception:
            return {}

    async def _get_watch_status(self) -> str:
        """D-06: Include Gmail Watch renewal status in digest header."""
        try:
            expiry_ts = await self.redis.get(WATCH_EXPIRY_REDIS_KEY)
            if expiry_ts:
                expiry_dt = datetime.fromisoformat(expiry_ts.decode())
                days_left = (expiry_dt - datetime.utcnow()).days
                if days_left <= 1:
                    return f"⚠️ Gmail Watch expires in {days_left}d — renew in Settings"
                return f"📡 Email watch: active ✅ (renews in {days_left}d)"
        except Exception:
            pass
        return "📡 Email watch: status unknown"

    async def _get_stalled_threads(self, restaurant_id: str) -> List[dict]:
        """Conversations awaiting reply >24h."""
        try:
            from datetime import timedelta
            cutoff = (datetime.utcnow() - timedelta(hours=24)).isoformat()
            result = (self.supabase.table("procurement_conversations")
                      .select("provider_id, message_text, created_at, providers!inner(name)")
                      .eq("restaurant_id", restaurant_id)
                      .eq("direction", "inbound")
                      .lt("created_at", cutoff)
                      .order("created_at", desc=True)
                      .limit(10).execute())
            return [
                {"provider_name": r.get("providers", {}).get("name", "Unknown"),
                 "snippet": r["message_text"][:80]}
                for r in (result.data or [])
            ]
        except Exception:
            return []

    async def _get_procurement_gaps(self, restaurant_id: str) -> List[dict]:
        """Calendar events in 7 days where wine stock < threshold."""
        try:
            from datetime import timedelta
            today = date.today()
            cutoff = (today + timedelta(days=7)).isoformat()
            events = (self.supabase.table("calendar_events")
                      .select("id, title, event_date")
                      .eq("restaurant_id", restaurant_id)
                      .in_("event_type", ["tasting", "tasting_event", "high_volume_expected"])
                      .gte("event_date", today.isoformat())
                      .lte("event_date", cutoff)
                      .execute())
            gaps = []
            for ev in (events.data or []):
                days_away = (date.fromisoformat(ev["event_date"]) - today).days
                gaps.append({
                    "event_title": ev["title"],
                    "days_away": days_away,
                    "wine_gap": "check inventory",
                })
            return gaps
        except Exception:
            return []
```
