"""
Model tier client factory for Phase 24.
Tier 1: Gemini 2.0 Flash (google-genai SDK) — every message triage
Tier 2+3: Claude Haiku 4.5 (anthropic SDK) — flag confirmation + summarization

CRITICAL: Use `from google import genai` (NOT `import google.generativeai`).
The old SDK does NOT support gemini-2.0-flash.
"""
import json
import logging
from typing import Any

import anthropic
from google import genai
from google.genai import types

from config.settings import Settings

logger = logging.getLogger(__name__)
_settings: Settings | None = None
_gemini_client: "GeminiFlashClient | None" = None
_haiku_client: "HaikuClient | None" = None


def _get_settings() -> Settings:
    global _settings
    if _settings is None:
        _settings = Settings()
    return _settings


class GeminiFlashClient:
    """Tier 1 model client — Gemini 2.0 Flash via google-genai SDK.
    
    Used for: every-message signal detection, flag classification, structured extraction.
    JSON mode enforced via response_mime_type='application/json'.
    """

    def __init__(self, api_key: str, model: str = "gemini-2.0-flash"):
        self._client = genai.Client(api_key=api_key)
        self._model = model

    async def generate_json(
        self,
        prompt: str,
        system_prompt: str | None = None,
        temperature: float = 0.1,
    ) -> dict | list:
        """Call Gemini with JSON mode enforced. Returns parsed dict/list.
        Never raises on JSON parse failure — returns {} on any error."""
        contents = prompt
        if system_prompt:
            contents = f"{system_prompt}\n\n{prompt}"
        try:
            response = await self._client.aio.models.generate_content(
                model=self._model,
                contents=contents,
                config=types.GenerateContentConfig(
                    response_mime_type="application/json",
                    temperature=temperature,
                ),
            )
            return json.loads(response.text)
        except json.JSONDecodeError as e:
            logger.warning(f"GeminiFlashClient JSON parse failed: {e}. Response: {getattr(response, 'text', '')[:200]}")
            return {}
        except Exception as e:
            logger.error(f"GeminiFlashClient.generate_json error: {e}", exc_info=True)
            return {}

    async def generate_text(self, prompt: str, temperature: float = 0.3) -> str:
        """Call Gemini for free-form text. Returns empty string on failure."""
        try:
            response = await self._client.aio.models.generate_content(
                model=self._model,
                contents=prompt,
                config=types.GenerateContentConfig(temperature=temperature),
            )
            return response.text or ""
        except Exception as e:
            logger.error(f"GeminiFlashClient.generate_text error: {e}", exc_info=True)
            return ""


class HaikuClient:
    """Tier 2 + Tier 3 model client — Claude Haiku 4.5 via anthropic SDK.
    
    Tier 2: flag confirmation (ambiguous signals, confidence < 0.7)
    Tier 3: session close deep summarization (max_tokens=1024)
    """

    def __init__(self, api_key: str, model: str = "claude-haiku-4-5-20251001"):
        self._client = anthropic.AsyncAnthropic(api_key=api_key)
        self._model = model

    async def complete(
        self,
        prompt: str,
        system_prompt: str | None = None,
        max_tokens: int = 512,
    ) -> str:
        """Call Haiku and return text response. Returns empty string on failure."""
        messages = [{"role": "user", "content": prompt}]
        kwargs: dict[str, Any] = {
            "model": self._model,
            "max_tokens": max_tokens,
            "messages": messages,
        }
        if system_prompt:
            kwargs["system"] = system_prompt
        try:
            response = await self._client.messages.create(**kwargs)
            return response.content[0].text
        except Exception as e:
            logger.error(f"HaikuClient.complete error: {e}", exc_info=True)
            return ""

    async def summarize(self, session_transcript: str, system_prompt: str) -> str:
        """Tier 3 summarization call — uses max_tokens=1024."""
        return await self.complete(
            prompt=session_transcript,
            system_prompt=system_prompt,
            max_tokens=1024,
        )


def get_gemini_client() -> GeminiFlashClient:
    """Singleton factory for Gemini 2.0 Flash client."""
    global _gemini_client
    if _gemini_client is None:
        s = _get_settings()
        _gemini_client = GeminiFlashClient(api_key=s.google_api_key, model=s.gemini_model)
    return _gemini_client


def get_haiku_client() -> HaikuClient:
    """Singleton factory for Claude Haiku 4.5 client."""
    global _haiku_client
    if _haiku_client is None:
        s = _get_settings()
        _haiku_client = HaikuClient(api_key=s.claude_api_key, model=s.haiku_model)
    return _haiku_client
