**Step 1 — Create apps/web/src/components/communications/ directory and ConversationComposer.tsx:**

The component is a slide-in panel that:
1. Shows the AI draft in a textarea (pre-filled)
2. Has "Why this reply?" collapsible section (D-09)
3. Has read-only To: field + optional CC/BCC inputs (D-10)
4. Has Send button + Cancel button

```tsx
import React, { useState } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import { ChevronDown, ChevronUp, X, Send, AlertCircle } from 'lucide-react';

interface ConversationComposerProps {
  isOpen: boolean;
  onClose: () => void;
  draft: {
    id: string;
    content: string;
    toEmail: string;  // read-only — provider primary email
    providerName: string;
    reasoning?: string;  // from decision_log
  };
  onApprove: (editedContent: string, cc?: string, bcc?: string) => Promise<void>;
  onReject: () => Promise<void>;
}

export const ConversationComposer: React.FC<ConversationComposerProps> = ({
  isOpen, onClose, draft, onApprove, onReject,
}) => {
  const [content, setContent] = useState(draft.content);
  const [cc, setCc] = useState('');
  const [bcc, setBcc] = useState('');
  const [showReasoning, setShowReasoning] = useState(false);  // D-09: collapsed by default
  const [isSending, setIsSending] = useState(false);

  const handleSend = async () => {
    setIsSending(true);
    try {
      await onApprove(content, cc || undefined, bcc || undefined);
      onClose();
    } finally {
      setIsSending(false);
    }
  };

  return (
    <AnimatePresence>
      {isOpen && (
        <motion.div className="fixed inset-0 z-50" initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}>
          <div className="fixed inset-0 bg-black/20 backdrop-blur-[4px]" onClick={onClose} />
          <motion.div
            className="fixed right-0 top-0 h-full w-[520px] bg-white shadow-2xl flex flex-col"
            initial={{ x: "100%" }} animate={{ x: 0 }} exit={{ x: "100%" }}
            transition={{ type: "spring", damping: 30 }}
          >
            {/* Header */}
            <div className="px-6 py-4 border-b border-gray-100 flex items-center justify-between">
              <div>
                <h2 className="font-semibold text-gray-900">Reply to {draft.providerName}</h2>
                <p className="text-xs text-gray-500 mt-0.5">Review before sending</p>
              </div>
              <button onClick={onClose} className="p-2 hover:bg-gray-50 rounded-lg">
                <X className="w-4 h-4 text-gray-400" />
              </button>
            </div>

            {/* Composer body */}
            <div className="flex-1 overflow-y-auto px-6 py-4 space-y-4">
              {/* To field — read-only per D-10 */}
              <div>
                <label className="text-xs font-medium text-gray-500 uppercase tracking-[0.08em]">To</label>
                <div className="mt-1 px-3 py-2 bg-gray-50 rounded-lg text-sm text-gray-700 border border-gray-200 flex items-center gap-2">
                  <span className="flex-1">{draft.toEmail}</span>
                  <span className="text-xs text-gray-400 flex items-center gap-1">
                    <AlertCircle className="w-3 h-3" /> locked
                  </span>
                </div>
              </div>

              {/* CC / BCC — D-10 */}
              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="text-xs font-medium text-gray-500 uppercase tracking-[0.08em]">CC</label>
                  <input
                    value={cc}
                    onChange={(e) => setCc(e.target.value)}
                    placeholder="cc@example.com"
                    className="mt-1 w-full px-3 py-2 border border-gray-200 rounded-lg text-sm"
                  />
                </div>
                <div>
                  <label className="text-xs font-medium text-gray-500 uppercase tracking-[0.08em]">BCC</label>
                  <input
                    value={bcc}
                    onChange={(e) => setBcc(e.target.value)}
                    placeholder="bcc@example.com"
                    className="mt-1 w-full px-3 py-2 border border-gray-200 rounded-lg text-sm"
                  />
                </div>
              </div>

              {/* Draft content */}
              <div>
                <label className="text-xs font-medium text-gray-500 uppercase tracking-[0.08em]">Message</label>
                <textarea
                  value={content}
                  onChange={(e) => setContent(e.target.value)}
                  rows={12}
                  className="mt-1 w-full px-3 py-2 border border-gray-200 rounded-lg text-sm resize-none focus:outline-none focus:ring-2 focus:ring-[#8b1a2f]/20"
                />
              </div>

              {/* Why this reply? — D-09, collapsed by default */}
              {draft.reasoning && (
                <div>
                  <button
                    onClick={() => setShowReasoning(!showReasoning)}
                    className="flex items-center gap-2 text-sm text-gray-500 hover:text-gray-700"
                  >
                    {showReasoning ? <ChevronUp className="w-3 h-3" /> : <ChevronDown className="w-3 h-3" />}
                    Why this reply?
                  </button>
                  <AnimatePresence>
                    {showReasoning && (
                      <motion.div
                        initial={{ height: 0, opacity: 0 }}
                        animate={{ height: "auto", opacity: 1 }}
                        exit={{ height: 0, opacity: 0 }}
                        className="overflow-hidden"
                      >
                        <div className="mt-2 p-3 bg-gray-50 rounded-md text-sm text-gray-600 leading-relaxed">
                          {draft.reasoning}
                        </div>
                      </motion.div>
                    )}
                  </AnimatePresence>
                </div>
              )}
            </div>

            {/* Footer actions */}
            <div className="px-6 py-4 border-t border-gray-100 flex items-center justify-between">
              <button
                onClick={onReject}
                className="px-4 py-2 text-sm text-gray-500 hover:text-red-600"
              >
                Reject
              </button>
              <button
                onClick={handleSend}
                disabled={isSending || !content.trim()}
                className="flex items-center gap-2 px-5 py-2 bg-[#8b1a2f] text-white text-sm font-medium rounded-lg hover:bg-[#701525] disabled:opacity-50"
              >
                <Send className="w-3.5 h-3.5" />
                {isSending ? 'Sending...' : 'Approve & Send'}
              </button>
            </div>
          </motion.div>
        </motion.div>
      )}
    </AnimatePresence>
  );
};

export default ConversationComposer;
```

```typescript
async sendDraftReminders(): Promise<void> {
  const twoHoursAgo = new Date(Date.now() - 2 * 60 * 60 * 1000).toISOString();
  const { data: staleDrafts } = await this.supabase
    .from('provider_conversation_sessions')
    .select('id, restaurant_id, provider_id, draft_created_at')
    .eq('session_status', 'pending_approval')
    .lt('draft_created_at', twoHoursAgo)
    .is('reminder_sent_at', null);

  for (const draft of staleDrafts || []) {
    await this.notificationsService.createNotification({
      restaurantId: draft.restaurant_id,
      type: 'draft_reminder',
      title: '⏳ Draft reply still waiting',
      message: 'A vendor reply draft has been awaiting approval for 2+ hours.',
      priority: 'medium',
      actionUrl: '/providers',
      metadata: { session_id: draft.id, provider_id: draft.provider_id },
    });
    await this.supabase
      .from('provider_conversation_sessions')
      .update({ reminder_sent_at: new Date().toISOString() })
      .eq('id', draft.id);
  }
}
```

**Method 2 — discardExpiredDrafts() (called hourly by scheduled-tasks.service.ts):**
```typescript
async discardExpiredDrafts(): Promise<void> {
  const fortyEightHoursAgo = new Date(Date.now() - 48 * 60 * 60 * 1000).toISOString();
  const { data: expiredDrafts } = await this.supabase
    .from('provider_conversation_sessions')
    .select('id, restaurant_id, provider_id')
    .eq('session_status', 'pending_approval')
    .lt('draft_created_at', fortyEightHoursAgo);

  for (const draft of expiredDrafts || []) {
    await this.supabase
      .from('provider_conversation_sessions')
      .update({
        session_status: 'closed',
        draft_discarded_at: new Date().toISOString(),
      })
      .eq('id', draft.id);
    // Log the expiry to decision_log (D-11)
    await this.supabase.from('decision_log').insert({
      agent_name: 'conversations_service',
      decision_type: 'draft_expired',
      inputs: { session_id: draft.id, provider_id: draft.provider_id },
      output: { action: 'auto_discarded' },
      reasoning: 'Draft not approved within 48 hours — auto-discarded per D-11',
      confidence: 1.0,
      restaurant_id: draft.restaurant_id,
    });
  }
}
```
