import asyncio
import hashlib
import json
import logging
from dataclasses import dataclass
from datetime import date, datetime, timezone, timedelta
from typing import Any

from core.base_agent import BaseAgent
from services.model_clients import get_gemini_client
from config.settings import Settings

logger = logging.getLogger(__name__)

CLASSIFICATION_PROMPT = """You are an email classification agent for a restaurant wine manager.
Classify the following email into exactly one category:
- OPERATIONAL: order status, invoice, delivery confirmation, account query, complaint, meeting request
- PROMO: discount offer, clearance sale, new product launch, exclusive allocation, seasonal promotion
- NOISE: spam, newsletter, automated notification with no action needed, out-of-office

Return JSON with exactly these fields:
{
  "classification": "OPERATIONAL" | "PROMO" | "NOISE",
  "confidence": 0.0-1.0,
  "reasoning": "brief reason"
}

Subject: {subject}
Body (first 500 chars): {body_preview}"""

PROMO_EXTRACTION_PROMPT = """Extract promotional details from this vendor email.
Return JSON with these fields (null if not found):
{
  "product_name": str | null,
  "grape_variety": str | null,
  "region": str | null,
  "discount_pct": float | null,
  "discount_fixed": float | null,
  "valid_from": "YYYY-MM-DD" | null,
  "valid_until": "YYYY-MM-DD" | null,
  "promo_description": str,
  "conditions": str | null,
  "min_quantity": int | null
}
Subject: {subject}
Body: {body}"""


class EmailIntelAgent(BaseAgent):
    """Email Intelligence Agent — classifies inbound emails before routing.
    
    OPERATIONAL → re-publish to EmailParsingAgent
    PROMO → extract → vendor_promotions → MenuFitAnalyzer → digest
    NOISE → discard
    """

    def __init__(self, settings: Settings):
        super().__init__(name="email_intel_agent", settings=settings)
        self._gemini = None  # lazy-loaded via get_gemini_client()

    async def initialize(self):
        await super().initialize()
        self._gemini = get_gemini_client()
        self.logger.info("EmailIntelAgent initialized — Tier 1 Gemini 2.0 Flash ready")

    async def process_message(self, routing_key: str, payload: dict[str, Any]) -> None:
        """Level 4 entry point — idempotency gated."""
        message_id = payload.get("gmail_message_id") or payload.get("message_id", "")
        correlation_id = payload.get("correlation_id")
        self._current_correlation_id = correlation_id

        if await self._check_idempotency(f"email_intel:{message_id}"):
            self.logger.info(f"Duplicate email {message_id} — skipping")
            return

        try:
            await self._route_message(routing_key, payload)
            await self._mark_processed(f"email_intel:{message_id}", {"status": "ok"})
        except Exception as e:
            self.logger.error(f"EmailIntelAgent failed: {e}", exc_info=True)
            await self._send_to_dlq(payload, str(e), routing_key)
            raise

    async def _route_message(self, routing_key: str, payload: dict[str, Any]) -> None:
        if routing_key == "email.inbound.received":
            await self._handle_inbound_email(payload)

    async def _handle_inbound_email(self, payload: dict[str, Any]) -> None:
        subject = payload.get("subject", "")
        body = payload.get("body", "") or payload.get("body_text", "")
        sender_email = payload.get("sender_email", "")
        restaurant_id = payload.get("restaurant_id", "")
        provider_id = payload.get("provider_id", "")

        classification_result = await self._classify_email(subject, body)
        classification = classification_result.get("classification", "NOISE")
        confidence = classification_result.get("confidence", 0.5)

        await self.log_decision(
            decision_type="email_classification",
            inputs={"subject": subject[:200], "sender": sender_email, "confidence": confidence},
            output={"classification": classification},
            reasoning=classification_result.get("reasoning", ""),
            confidence=confidence,
        )

        if classification == "OPERATIONAL":
            # Re-publish for EmailParsingAgent — payload is unchanged
            await self.publish("email.events", "email.inbound.received", {
                **payload,
                "intel_classified": True,
                "intel_classification": "OPERATIONAL",
            })
            self.logger.info(f"OPERATIONAL email forwarded: {subject[:60]}")

        elif classification == "PROMO":
            await self._handle_promo_email(payload, subject, body, provider_id, restaurant_id)

        else:  # NOISE
            self.logger.debug(f"NOISE email discarded: {subject[:60]}")

    async def _classify_email(self, subject: str, body: str) -> dict:
        prompt = CLASSIFICATION_PROMPT.format(
            subject=subject,
            body_preview=body[:500],
        )
        result = await self._gemini.generate_json(prompt)
        return result if result else {"classification": "NOISE", "confidence": 0.5}

    async def _handle_promo_email(
        self, payload: dict, subject: str, body: str,
        provider_id: str, restaurant_id: str,
    ) -> None:
        extracted = await self._extract_promo(subject, body)
        if not extracted:
            return

        # Generate dedup_hash = SHA256(provider_id + product_name + discount_pct + valid_until)
        dedup_input = f"{provider_id}:{extracted.get('product_name','')}" \
                      f":{extracted.get('discount_pct','')}" \
                      f":{extracted.get('valid_until','')}"
        dedup_hash = hashlib.sha256(dedup_input.encode()).hexdigest()[:32]

        if await self._check_dedup(provider_id, dedup_hash, restaurant_id):
            self.logger.info(f"Promo suppressed (dedup): {subject[:60]}")
            return

        # Insert vendor_promotions row
        promo_row = {
            "provider_id": provider_id,
            "restaurant_id": restaurant_id,
            "detected_from_email_subject": subject[:250],
            "dedup_hash": dedup_hash,
            "status": "active",
            **{k: v for k, v in extracted.items() if v is not None},
        }
        conv_id = payload.get("detected_from_conversation_id")
        if conv_id:
            promo_row["detected_from_conversation_id"] = conv_id

        result = self.supabase.table("vendor_promotions").insert(promo_row).execute()
        promo_id = result.data[0]["id"] if result.data else None

        self.logger.info(f"Promo stored: {extracted.get('product_name')} {extracted.get('discount_pct')}% off")

        # Accumulate in daily digest Redis key
        today = date.today().isoformat()
        digest_key = f"digest:{restaurant_id}:{today}"
        await self.redis.lpush(digest_key, json.dumps({
            "promo_id": promo_id,
            "product_name": extracted.get("product_name"),
            "discount_pct": extracted.get("discount_pct"),
            "valid_until": extracted.get("valid_until"),
            "promo_description": extracted.get("promo_description", ""),
        }))
        await self.redis.expire(digest_key, 86400 * 2)  # 48h TTL

        # Trigger MenuFitAnalyzer check — lazy import to avoid circular deps
        from services.menu_fit_analyzer import MenuFitAnalyzer
        analyzer = MenuFitAnalyzer(supabase=self.supabase, gemini=self._gemini)
        fit_result = await analyzer.check_fit(
            product_name=extracted.get("product_name", ""),
            grape_variety=extracted.get("grape_variety"),
            region=extracted.get("region"),
            restaurant_id=restaurant_id,
        )

        # Update vendor_promotions with menu_fit result
        if promo_id and fit_result:
            self.supabase.table("vendor_promotions").update({
                "menu_fit": fit_result.fit,
                "menu_fit_detail": fit_result.detail,
            }).eq("id", promo_id).execute()

            # PROCUREMENT_NUDGE flag if fit is not NO_FIT (INTEL-04)
            if fit_result.fit in ("STRONG_FIT", "PARTIAL_FIT"):
                await self.publish("agent.events", "agent.flag.created", {
                    "provider_id": provider_id,
                    "restaurant_id": restaurant_id,
                    "flag_type": "PROMO_DETECTED",
                    "action_type": "PROCUREMENT_NUDGE",
                    "urgency_tier": "STANDARD",
                    "title": f"New promo: {extracted.get('product_name')} — {fit_result.fit}",
                    "description": f"{extracted.get('discount_pct')}% off. {fit_result.detail}",
                    "suggested_action": "Review deal and check current stock",
                    "relevance_score": 0.75 if fit_result.fit == "STRONG_FIT" else 0.55,
                })

    async def _extract_promo(self, subject: str, body: str) -> dict | None:
        prompt = PROMO_EXTRACTION_PROMPT.format(subject=subject, body=body[:1000])
        result = await self._gemini.generate_json(prompt)
        return result if result and result.get("promo_description") else None

    async def _check_dedup(self, provider_id: str, dedup_hash: str, restaurant_id: str) -> bool:
        """Returns True if same dedup_hash found active within 7 days."""
        try:
            cutoff = (datetime.now(timezone.utc) - timedelta(days=7)).isoformat()
            result = (
                self.supabase.table("vendor_promotions")
                .select("id")
                .eq("provider_id", provider_id)
                .eq("restaurant_id", restaurant_id)
                .eq("dedup_hash", dedup_hash)
                .eq("status", "active")
                .gte("created_at", cutoff)
                .limit(1)
                .execute()
            )
            return bool(result.data)
        except Exception as e:
            self.logger.warning(f"Dedup check failed: {e} — allowing through")
            return False  # fail open

"""
MenuFitAnalyzer — per D-02 (locked decision).
Queries current wine list from master_wine_library_submissions/master_wine_library,
runs LLM comparison against promotion product.
Returns STRONG_FIT / PARTIAL_FIT / NO_FIT.
"""
import logging
from dataclasses import dataclass
from typing import Any

logger = logging.getLogger(__name__)

MENU_FIT_PROMPT = """You are analyzing whether a wine promotion fits the current restaurant menu.

Current menu wines (names and types):
{menu_wines}

Promotion being checked:
- Product: {product_name}
- Grape variety: {grape_variety}
- Region: {region}

Assess menu fit and return JSON:
{{
  "fit": "STRONG_FIT" | "PARTIAL_FIT" | "NO_FIT",
  "matching_count": int,
  "detail": "Matches N current menu items / Similar category / Not in current program",
  "matching_items": ["wine name 1", "wine name 2"]
}}

STRONG_FIT: Same grape/type as 1+ menu items
PARTIAL_FIT: Similar category (e.g., same color/region) but different grape
NO_FIT: No overlap with current menu
"""


@dataclass
class MenuFitResult:
    fit: str  # STRONG_FIT | PARTIAL_FIT | NO_FIT
    detail: str
    matching_count: int = 0
    matching_items: list = None

    def __post_init__(self):
        if self.matching_items is None:
            self.matching_items = []


_NO_FIT_FALLBACK = MenuFitResult(fit="NO_FIT", detail="Menu check unavailable")


class MenuFitAnalyzer:
    """Queries current wine list and uses Gemini Flash to assess promotion fit.
    Per D-02: Returns STRONG_FIT / PARTIAL_FIT / NO_FIT.
    """

    def __init__(self, supabase: Any, gemini: Any):
        self._supabase = supabase
        self._gemini = gemini

    async def check_fit(
        self,
        product_name: str,
        grape_variety: str | None,
        region: str | None,
        restaurant_id: str,
    ) -> MenuFitResult:
        """Check if a promo product fits the current restaurant wine menu."""
        try:
            menu_wines = await self._get_current_menu(restaurant_id)
            if not menu_wines:
                return MenuFitResult(fit="NO_FIT", detail="No wines on current menu")

            prompt = MENU_FIT_PROMPT.format(
                menu_wines="\n".join(f"- {w}" for w in menu_wines[:50]),  # cap at 50
                product_name=product_name or "Unknown",
                grape_variety=grape_variety or "Unknown",
                region=region or "Unknown",
            )
            result = await self._gemini.generate_json(prompt)
            if not result or "fit" not in result:
                return _NO_FIT_FALLBACK

            return MenuFitResult(
                fit=result.get("fit", "NO_FIT"),
                detail=result.get("detail", ""),
                matching_count=result.get("matching_count", 0),
                matching_items=result.get("matching_items", []),
            )
        except Exception as e:
            logger.warning(f"MenuFitAnalyzer.check_fit failed: {e}")
            return _NO_FIT_FALLBACK

    async def _get_current_menu(self, restaurant_id: str) -> list[str]:
        """Query master_wine_library_submissions for active wines on this restaurant's menu."""
        try:
            result = (
                self._supabase.table("master_wine_library_submissions")
                .select("wine_name, primary_type, grape_variety")
                .eq("restaurant_id", restaurant_id)
                .eq("status", "approved")
                .limit(100)
                .execute()
            )
            if not result.data:
                return []
            return [
                f"{w.get('wine_name', '')} ({w.get('primary_type', '')} {w.get('grape_variety', '')})"
                for w in result.data
            ]
        except Exception as e:
            logger.warning(f"MenuFitAnalyzer._get_current_menu failed: {e}")
            return []

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Inbound email → LLM prompt | Provider email body is injected verbatim into classification prompt |
| LLM response → DB write | Extracted promo data written to vendor_promotions without sanitization |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-24-04-01 | Tampering | Email body in classification prompt | mitigate | Body truncated to 500 chars; JSON mode enforced; output parsed via json.loads() — injection cannot escape structured output |
| T-24-04-02 | Tampering | Promo extraction body limit | mitigate | body[:1000] cap in PROMO_EXTRACTION_PROMPT — prevents unbounded context injection |
| T-24-04-03 | Spoofing | Dedup hash collision | accept | SHA256 32-char truncation; collision probability negligible at MVP scale; fails open (allows through) |
| T-24-04-04 | Information Disclosure | Redis digest key includes restaurant_id | accept | Redis key pattern `digest:{restaurant_id}:{date}` scoped per restaurant; Upstash TLS in production |
| T-24-04-05 | Denial of Service | PROMO flood from single vendor | mitigate | 7-day dedup suppression blocks re-alert; daily digest caps notification volume |
</threat_model>
