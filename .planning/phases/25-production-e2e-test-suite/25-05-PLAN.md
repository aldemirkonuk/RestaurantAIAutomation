        prod_supabase, restaurant_id: str, timeout_seconds: float = 30.0
    ) -> dict | None:
        """Poll notification_deliveries for email delivery within 30s (TEST-PROD-05)."""
        deadline = asyncio.get_event_loop().time() + timeout_seconds
        while asyncio.get_event_loop().time() < deadline:
            try:
                result = (
                    prod_supabase.table(DELIVERIES_TABLE)
                    .select("id, channel, status, created_at")
                    .eq("restaurant_id", restaurant_id)
                    .eq("channel", "email")
                    .order("created_at", desc=True)
                    .limit(1)
                    .execute()
                )
                if result.data and len(result.data) > 0:
                    row = result.data[0]
                    if row.get("status") in ("sent", "delivered", "queued"):
                        return row
            except Exception as exc:
                # Table may not exist yet or columns differ — log and continue polling
                _ = exc  # suppress lint warning
            await asyncio.sleep(3.0)
        return None
