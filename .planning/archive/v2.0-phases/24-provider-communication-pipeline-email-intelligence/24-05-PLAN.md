# SENSITIVE_TOPIC_SIGNALS from RESEARCH.md Pattern 8
SENSITIVE_TOPIC_SIGNALS = [
    "health", "medical", "surgery", "hospital", "diagnosis",
    "grief", "loss", "passed away", "funeral", "bereavement",
    "financial trouble", "bankruptcy", "debt problem",
    "marriage problem", "divorce", "separation",
    "personal", "private", "confidential", "between us",
]

def _classify_message_sensitivity(self, message: str) -> bool:
    """ETHC-01, D-12: Returns True if message contains sensitive personal content.
    Fast keyword pre-filter — conservative: marks sensitive if any signal keyword present.
    Independent of close-relationship mode (D-14)."""
    lowered = message.lower()

def _build_draft_system_prompt(
    self, base_system_prompt: str, relationship_posture: str, manager_instructions: list
) -> str:
    """D-13: Apply close-relationship posture to draft message system prompt."""
    prompt = base_system_prompt
    if relationship_posture == "collaborative":

    return prompt
