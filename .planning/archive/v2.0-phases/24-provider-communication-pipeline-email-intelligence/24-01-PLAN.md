Also add the Supabase RPC function for semantic search (used by agent at line 958 via `supabase.rpc("match_conversation_embeddings", ...)`):
```sql
CREATE OR REPLACE FUNCTION match_conversation_embeddings(
    query_embedding VECTOR(768),
    match_provider_id UUID,
    match_restaurant_id UUID,
    match_threshold FLOAT DEFAULT 0.5,
    match_count INT DEFAULT 10
)
RETURNS TABLE (
    id UUID, message_text TEXT, role VARCHAR, session_id UUID,
    importance_score FLOAT, created_at TIMESTAMPTZ, similarity FLOAT
)
LANGUAGE SQL STABLE AS $$
    SELECT id, message_text, role, session_id, importance_score, created_at,
           1 - (embedding <=> query_embedding) AS similarity
    FROM conversation_embeddings
    WHERE provider_id = match_provider_id
      AND restaurant_id = match_restaurant_id
      AND has_signal = true
      AND sensitive = false
      AND 1 - (embedding <=> query_embedding) > match_threshold
    ORDER BY embedding <=> query_embedding
    LIMIT match_count;
$$;
```
