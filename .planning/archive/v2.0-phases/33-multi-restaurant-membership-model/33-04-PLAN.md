**Step 1 — Create `apps/web/src/pages/InviteLanding.tsx`**:

This page handles 4 states per UI-SPEC.md Surface 1:
- State A: Logged-out + valid code → show two buttons (sign in / create account)
- State B: Logged-in + valid code + not yet a member → show confirm button
- State C: Expired/invalid/used code → show expired message
- State D: Already a member → toast + redirect (handled via 409 response from backend)

```typescript
import { useState, useEffect } from 'react'
import { useNavigate, useParams, Link } from 'react-router-dom'
import { motion } from 'framer-motion'
import { useAuth } from '../contexts/AuthContext'
import { Wine, AlertCircle, RefreshCw } from 'lucide-react'
import { toast } from 'sonner'

const API_URL = import.meta.env.VITE_API_GATEWAY_URL || 'http://localhost:4000'

interface InvitePreview {
  valid: boolean
  reason?: string
  restaurant?: string
  city?: string
  inviter?: string
  role?: string
}

// Role badge colors per UI-SPEC.md Color section
function RoleBadge({ role }: { role: string }) {
  const colorMap: Record<string, string> = {
    owner: 'bg-wine-50 text-wine-600 ring-wine-100',
    manager: 'bg-blue-50 text-blue-600 ring-blue-100',
    staff: 'bg-gray-100 text-gray-600 ring-gray-200',
  }
  return (
    <span className={`text-xs font-medium px-2 py-1 rounded-full ring-1 capitalize ${colorMap[role] ?? colorMap.staff}`}>
      {role}
    </span>
  )
}

export function InviteLanding() {
  const { code } = useParams<{ code: string }>()
  const navigate = useNavigate()
  const { user, refreshBranches } = useAuth()

  const [preview, setPreview] = useState<InvitePreview | null>(null)
  const [loadingPreview, setLoadingPreview] = useState(true)
  const [submitting, setSubmitting] = useState(false)
  const [error, setError] = useState<string | null>(null)

  // Fetch invite preview on mount (public endpoint — no auth needed)
  useEffect(() => {
    if (!code) return
    setLoadingPreview(true)
    fetch(`${API_URL}/api/v1/auth/invite/${code.toUpperCase()}`)
      .then(r => r.json())
      .then(data => setPreview(data))
      .catch(() => setPreview({ valid: false, reason: 'network_error' }))
      .finally(() => setLoadingPreview(false))
  }, [code])

  const handleAccept = async () => {
    setSubmitting(true)
    setError(null)
    try {
      const token = localStorage.getItem('accessToken')
      const resp = await fetch(`${API_URL}/api/v1/auth/invite/${code}/accept`, {
        method: 'POST',
        headers: { Authorization: `Bearer ${token}` },
      })
      if (!resp.ok) {
        const data = await resp.json()
        if (data.message === 'already_member' || resp.status === 409) {
          // D-03: already a member — toast + redirect
          toast.success(`You're already a member of ${preview?.restaurant ?? 'this restaurant'}`)
          navigate('/')
          return
        }
        throw new Error(data.message || 'Failed to accept invite')
      }
      // Success: refresh branches so the new restaurant appears in branch switcher
      await refreshBranches?.()
      toast.success(`You've joined ${preview?.restaurant ?? 'the restaurant'}!`)
      navigate('/')
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : 'Failed to accept invite'
      setError(`Couldn't add you to ${preview?.restaurant ?? 'this restaurant'}. Please try again or contact the owner.`)
      console.error('acceptInvite error:', msg)
    } finally {
      setSubmitting(false)
    }
  }

  if (loadingPreview) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-slate-50">
        <RefreshCw className="w-6 h-6 text-gray-300 animate-spin" />
      </div>
    )
  }

  // State C: expired or invalid code
  if (!preview?.valid) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-slate-50 px-4 py-12">
        <motion.div
          initial={{ opacity: 0, y: 8 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ duration: 0.2 }}
          className="w-full max-w-md"
        >
          {/* WineOps branding */}
          <div className="text-center mb-8">
            <div className="inline-flex items-center justify-center w-16 h-16 bg-wine-600 rounded-2xl mb-4 shadow-lg">
              <Wine className="w-8 h-8 text-white" />
            </div>
          </div>
          <div className="bg-white rounded-2xl border border-gray-100 shadow-sm p-8 text-center">
            <AlertCircle className="w-8 h-8 text-rose-400 mx-auto mb-3" />
            <h1 className="text-lg font-semibold text-gray-900">This invite has expired</h1>
            <p className="text-sm text-gray-500 mt-2">
              This invite link is no longer valid. Ask the restaurant owner for a new one.
            </p>
            <Link
              to="/login"
              className="mt-6 inline-block border border-gray-200 text-gray-700 rounded-xl px-6 py-3 text-sm font-medium hover:bg-gray-50 transition-colors"
            >
              Back to sign in
            </Link>
          </div>
        </motion.div>
      </div>
    )
  }

  return (
    <div className="min-h-screen flex items-center justify-center bg-slate-50 px-4 py-12">
      <motion.div
        initial={{ opacity: 0, y: 8 }}
        animate={{ opacity: 1, y: 0 }}
        transition={{ duration: 0.2 }}
        className="w-full max-w-md"
      >
        {/* WineOps branding */}
        <div className="text-center mb-8">
          <div className="inline-flex items-center justify-center w-16 h-16 bg-wine-600 rounded-2xl mb-4 shadow-lg">
            <Wine className="w-8 h-8 text-white" />
          </div>
        </div>

        <div className="bg-white rounded-2xl border border-gray-100 shadow-sm p-8">
          {/* Error display */}
          {error && (
            <div className="mb-4 p-3 bg-red-50 border border-red-200 rounded-lg flex items-center gap-2">
              <AlertCircle className="w-4 h-4 text-red-500 flex-shrink-0" />
              <p className="text-sm text-red-700">{error}</p>
            </div>
          )}

          {user ? (
            // State B: logged-in user
            <>
              <h1 className="text-lg font-semibold text-gray-900 mb-2">
                Add {preview.restaurant} to your branches?
              </h1>
              <p className="text-sm text-gray-500 mb-3">
                You'll be added as <strong className="capitalize">{preview.role}</strong> at {preview.restaurant}.
              </p>
              {preview.role && <RoleBadge role={preview.role} />}
              <div className="mt-6 space-y-3">
                <button
                  onClick={handleAccept}
                  disabled={submitting}
                  className="w-full py-3 bg-wine-600 hover:bg-wine-700 disabled:opacity-60 text-white text-sm font-medium rounded-xl transition-colors flex items-center justify-center gap-2"
                >
                  {submitting ? (
                    <>
                      <RefreshCw className="w-4 h-4 animate-spin" />
                      Adding…
                    </>
                  ) : (
                    `Add ${preview.restaurant}`
                  )}
                </button>
                <button
                  onClick={() => navigate(-1)}
                  className="w-full text-sm text-gray-400 underline hover:text-gray-600 transition-colors"
                >
                  Cancel, go back
                </button>
              </div>
            </>
          ) : (
            // State A: logged-out user
            <>
              <h1 className="text-lg font-semibold text-gray-900 mb-2">
                You've been invited to {preview.restaurant}
              </h1>
              <p className="text-sm text-gray-500 mb-1">
                Join {preview.restaurant} and start managing together.
              </p>
              {preview.inviter && (
                <p className="text-sm text-gray-400 mb-3">
                  Invited by <strong>{preview.inviter}</strong>
                </p>
              )}
              {preview.role && <RoleBadge role={preview.role} />}
              <div className="mt-6 space-y-3">
                <Link
                  to={`/login?redirect=/invite/${code}`}
                  className="w-full block text-center py-3 bg-wine-600 hover:bg-wine-700 text-white text-sm font-medium rounded-xl transition-colors"
                >
                  Sign in to accept
                </Link>
                <Link
                  to={`/register?invite=${code}`}
                  className="w-full block text-center py-3 border border-gray-200 text-gray-700 text-sm font-medium rounded-xl hover:bg-gray-50 transition-colors"
                >
                  Create account to accept
                </Link>
              </div>
              {preview.inviter && (
                <p className="text-xs text-gray-400 text-center mt-4">
                  Invite expires in 7 days · Single use
                </p>
              )}
            </>
          )}
        </div>
      </motion.div>
    </div>
  )
}

export default InviteLanding
```

**Step 2 — Create `apps/web/src/pages/NoAccess.tsx`**:

```typescript
import { useNavigate } from 'react-router-dom'
import { motion } from 'framer-motion'
import { useAuth } from '../contexts/AuthContext'
import { UserX, Wine } from 'lucide-react'
import { useSearchParams } from 'react-router-dom'

export function NoAccess() {
  const navigate = useNavigate()
  const [searchParams] = useSearchParams()
  const restaurantName = searchParams.get('from') ?? 'this location'
  const { availableRestaurants, setActiveRestaurantId } = useAuth()

  const otherBranches = availableRestaurants ?? []

  const handleSwitch = async (restaurantId: string) => {
    await setActiveRestaurantId(restaurantId)
    navigate('/')
  }

  return (
    <div className="min-h-screen flex items-center justify-center bg-slate-50 px-4">
      <motion.div
        initial={{ opacity: 0, y: 12 }}
        animate={{ opacity: 1, y: 0 }}
        transition={{ duration: 0.25 }}
        className="w-full max-w-md"
      >
        {/* Branding */}
        <div className="text-center mb-8">
          <div className="inline-flex items-center justify-center w-16 h-16 bg-wine-600 rounded-2xl mb-4 shadow-lg">
            <Wine className="w-8 h-8 text-white" />
          </div>
        </div>

        <div className="bg-white rounded-2xl border border-gray-100 shadow-sm p-8 text-center">
          <UserX className="w-10 h-10 text-gray-300 mx-auto mb-4" />
          <h1 className="text-lg font-semibold text-gray-900">
            You no longer have access to {restaurantName}
          </h1>
          <p className="text-sm text-gray-500 mt-2">
            Your access to this location has been removed.
          </p>

          {otherBranches.length > 0 ? (
            <div className="mt-6">
              <p className="text-xs text-gray-400 mb-3">Switch to another location:</p>
              <div className="space-y-2">
                {otherBranches.map((branch: any) => (
                  <button
                    key={branch.id}
                    onClick={() => handleSwitch(branch.id)}
                    className="w-full flex items-center gap-2 px-5 py-3 bg-wine-600 hover:bg-wine-700 text-white text-sm font-medium rounded-xl transition-colors"
                  >
                    Switch to {branch.name}
                  </button>
                ))}
              </div>
            </div>
          ) : (
            <a
              href="/login"
              className="mt-4 inline-block text-sm text-gray-400 hover:text-gray-600 underline"
            >
              Sign in to a different account
            </a>
          )}
        </div>
      </motion.div>
    </div>
  )
}

export default NoAccess
```
