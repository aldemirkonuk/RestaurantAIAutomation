```typescript
async joinViaInvite(dto: JoinViaInviteDto): Promise<TokenPair> {
  // Step 1: Consume invite atomically (unchanged — prevents TOCTOU race)
  const { data: invite, error: inviteErr } = await this.databaseService.supabase
    .from('organization_invites')
    .update({ used_at: new Date().toISOString(), used_by_email: dto.email })
    .eq('code', dto.code.toUpperCase())
    .is('used_at', null)
    .gt('expires_at', new Date().toISOString())
    .select('id, organization_id, restaurant_id, role')
    .single();

  if (inviteErr || !invite) {
    throw new BadRequestException('Invite code is invalid, expired, or already used');
  }

  // Step 2: Dual-path — existing user vs new user
  const { data: existingUser } = await this.databaseService.supabase
    .from('users')
    .select('*')
    .eq('email', dto.email)
    .maybeSingle();

  let user: any;

  if (existingUser) {
    // RC-1 fix: existing user joins a new restaurant — do NOT create duplicate user
    // Check for duplicate membership (CONTEXT D-03)
    const { data: existingAccess } = await this.databaseService.supabase
      .from('user_restaurant_access')
      .select('id')
      .eq('user_id', existingUser.user_id)
      .eq('restaurant_id', invite.restaurant_id)
      .maybeSingle();

    if (existingAccess) {
      // D-03: silent skip — un-consume invite so code can be re-used
      await this.databaseService.supabase
        .from('organization_invites')
        .update({ used_at: null, used_by_email: null })
        .eq('id', invite.id);
      throw new ConflictException('already_member');
    }

    user = existingUser;
  } else {
    // New user path (unchanged from original, minus the now-removed email check)
    const passwordHash = await bcrypt.hash(dto.password, this.SALT_ROUNDS);
    const { data: newUser, error: userErr } = await this.databaseService.supabase
      .from('users')
      .insert({
        email: dto.email,
        password_hash: passwordHash,
        name: dto.name,
        restaurant_id: invite.restaurant_id,
        role: invite.role,
        email_verified: true,
      })
      .select()
      .single();

    if (userErr || !newUser) {
      // Rollback invite consumption
      await this.databaseService.supabase
        .from('organization_invites')
        .update({ used_at: null, used_by_email: null })
        .eq('id', invite.id);
      throw new BadRequestException('User creation failed: ' + userErr?.message);
    }
    user = newUser;
  }

  // Step 3: Write user_restaurant_access row (RC-2 fix — both paths)
  await this.databaseService.supabase.from('user_restaurant_access').insert({
    user_id: user.user_id,
    restaurant_id: invite.restaurant_id,
    role: invite.role,
    invited_via: invite.id,
    is_active: true,
  });

  // Step 4: Upsert organization_members (handles both new and existing org members)
  await this.databaseService.supabase.from('organization_members').upsert(
    {
      organization_id: invite.organization_id,
      user_id: user.user_id,
      role: invite.role,
      invited_via: invite.id,
    },
    { onConflict: 'organization_id,user_id' },
  );

  return this.generateTokens({ ...user, restaurant_id: invite.restaurant_id });
}
```

```typescript
/**
 * Accept an invite as an already-authenticated user (no new account creation).
 * Called by POST /invite/:code/accept — handles existing-user and OAuth accounts.
 */
async acceptInviteAsExistingUser(userId: string, code: string): Promise<{ restaurant: string; role: string }> {
  // Validate and consume invite atomically
  const { data: invite, error: inviteErr } = await this.databaseService.supabase
    .from('organization_invites')
    .update({ used_at: new Date().toISOString(), used_by_email: userId })
    .eq('code', code.toUpperCase())
    .is('used_at', null)
    .gt('expires_at', new Date().toISOString())
    .select('id, organization_id, restaurant_id, role, restaurants(name)')
    .single();

  if (inviteErr || !invite) {
    throw new BadRequestException('Invite code is invalid, expired, or already used');
  }

  // Check for duplicate membership (D-03)
  const { data: existingAccess } = await this.databaseService.supabase
    .from('user_restaurant_access')
    .select('id')
    .eq('user_id', userId)
    .eq('restaurant_id', invite.restaurant_id)
    .maybeSingle();

  if (existingAccess) {
    // Un-consume invite; throw 409 (frontend maps to "already member" toast)
    await this.databaseService.supabase
      .from('organization_invites')
      .update({ used_at: null, used_by_email: null })
      .eq('id', invite.id);
    throw new ConflictException('already_member');
  }

  // Write URA row (works for OAuth users — no password involved)
  await this.databaseService.supabase.from('user_restaurant_access').insert({
    user_id: userId,
    restaurant_id: invite.restaurant_id,
    role: invite.role,
    invited_via: invite.id,
    is_active: true,
  });

  // Upsert organization_members
  await this.databaseService.supabase.from('organization_members').upsert(
    { organization_id: invite.organization_id, user_id: userId, role: invite.role },
    { onConflict: 'organization_id,user_id' },
  );

  const restaurantName = (invite.restaurants as any)?.name ?? 'restaurant';
  return { restaurant: restaurantName, role: invite.role };
}

/**
 * Get caller's role at a specific restaurant from user_restaurant_access.
 * Returns null if no URA row exists (caller should fall back to JWT role).
 */
async getUserRoleAtRestaurant(userId: string, restaurantId: string): Promise<string | null> {
  const { data } = await this.databaseService.supabase
    .from('user_restaurant_access')
    .select('role')
    .eq('user_id', userId)
    .eq('restaurant_id', restaurantId)
    .eq('is_active', true)
    .maybeSingle();
  return data?.role ?? null;
}
```

```typescript
 * Get the caller's role at a specific restaurant.
 * Called by frontend AuthContext on every branch switch (CONTEXT D-11).
 * Returns { role } from user_restaurant_access; null if no URA row exists (legacy user).
 */
@Get('me/role')
@UseGuards(JwtAuthGuard)
async getMyRole(
  @Req() req: Request & { user: any },
  @Query('restaurantId') restaurantId: string,

  if (!restaurantId) {
    return { success: true, role: null };

  const role = await this.authService.getUserRoleAtRestaurant(
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-33-02-01 | Spoofing | joinViaInvite invite code replay | mitigate | Atomic UPDATE WHERE used_at IS NULL prevents concurrent consumption; ConflictException returned if URA already exists for this restaurant |
| T-33-02-02 | Spoofing | acceptInviteAsExistingUser — attacker uses another user's JWT | mitigate | JwtAuthGuard validates Bearer token; userId comes from validated JWT payload, not request body |

| T-33-02-05 | Repudiation | already_member invite un-consumption | accept | Un-consuming invite on ConflictException allows code reuse — this is intentional (graceful handling, invite code value is low risk) |
</threat_model>
