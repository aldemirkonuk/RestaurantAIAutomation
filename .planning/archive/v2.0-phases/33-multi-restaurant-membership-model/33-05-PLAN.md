```typescript
  }
```

**Step 3 — Add role change handler**:

```typescript
const handleRoleChange = async (targetUserId: string, newRole: string) => {
  setLastOwnerWarning(null)
  setRoleUpdating(targetUserId)
  try {
    const token = localStorage.getItem('accessToken')
    const resp = await fetch(
      `${API_URL}/api/v1/restaurants/${activeRestaurantId}/members/${targetUserId}`,
      {
        method: 'PATCH',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${token}`,
        },
        body: JSON.stringify({ role: newRole }),
      },
    )
    if (!resp.ok) {
      const data = await resp.json()
      if (data.message?.includes('only owner')) {
        setLastOwnerWarning(targetUserId)
        return
      }
      throw new Error(data.message || 'Failed to update role')
    }
    const memberName = members.find(m => m.users?.user_id === targetUserId)?.users?.name ?? 'Member'
    toast.success(`${memberName} is now a ${newRole} at ${activeBranchName}`)
    await fetchMembers()
  } catch (err: unknown) {
    const msg = err instanceof Error ? err.message : 'Unknown error'
    toast.error(`Couldn't update role. ${msg}`)
  } finally {
    setRoleUpdating(null)
  }
}
```

**Step 4 — Add member removal handler**:

```typescript
const doRemove = async (targetUserId: string) => {
  setRemovingMemberId(targetUserId)
  try {
    const token = localStorage.getItem('accessToken')
    const resp = await fetch(
      `${API_URL}/api/v1/restaurants/${activeRestaurantId}/members/${targetUserId}`,
      {
        method: 'DELETE',
        headers: { Authorization: `Bearer ${token}` },
      },
    )
    if (!resp.ok) {
      const data = await resp.json()
      if (data.message?.includes('only owner')) {
        setLastOwnerWarning(targetUserId)
        setMemberToRemove(null)
        return
      }
      throw new Error(data.message || 'Failed to remove member')
    }
    const memberName = memberToRemove?.users?.name ?? 'Member'
    const isself = targetUserId === user?.userId
    toast.success(isself ? `You've left ${activeBranchName}` : `${memberName} has been removed from ${activeBranchName}`)
    setMemberToRemove(null)
    await fetchMembers()
  } catch (err: unknown) {
    const msg = err instanceof Error ? err.message : 'Unknown error'
    toast.error(`Failed to remove member. ${msg}`)
    setMemberToRemove(null)
  } finally {
    setRemovingMemberId(null)
  }
}
```

```typescript
const handleRevokeInvite = async (code: string) => {
  try {
    const token = localStorage.getItem('accessToken')
    const resp = await fetch(
      `${API_URL}/api/v1/restaurants/${activeRestaurantId}/invites/${code}`,
      { method: 'DELETE', headers: { Authorization: `Bearer ${token}` } },
    )
    if (!resp.ok) throw new Error('Revoke failed')
    toast.success('Invite revoked')
    setPendingInvites(prev => prev.filter(inv => inv.code !== code))
  } catch {
    toast.error("Couldn't revoke invite. Please try again.")
  }
}
```
