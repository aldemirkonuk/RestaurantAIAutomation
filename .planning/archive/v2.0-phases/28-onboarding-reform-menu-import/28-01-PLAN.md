# Phase 28 — Plan 01: Database — restaurant_menus + menu_items + user_onboarding_progress

## Objective
Create the three new tables that back the menu import feature and the activation checklist.

## Tasks

### Task 1: restaurant_menus migration
```sql
CREATE TABLE IF NOT EXISTS restaurant_menus (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  restaurant_id UUID NOT NULL REFERENCES restaurants(id) ON DELETE CASCADE,
  name          TEXT NOT NULL DEFAULT 'Wine List',
  type          TEXT NOT NULL DEFAULT 'beverage'
                CHECK (type IN ('beverage', 'food', 'full')),
  is_active     BOOLEAN NOT NULL DEFAULT TRUE,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX IF NOT EXISTS idx_restaurant_menus_restaurant ON restaurant_menus (restaurant_id);
ALTER TABLE restaurant_menus ENABLE ROW LEVEL SECURITY;
CREATE POLICY "menus_restaurant_isolation" ON restaurant_menus
  FOR ALL USING (restaurant_id = (SELECT restaurant_id FROM users WHERE id = auth.uid()));
```

### Task 2: menu_items migration
```sql
CREATE TABLE IF NOT EXISTS menu_items (
  id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  menu_id             UUID NOT NULL REFERENCES restaurant_menus(id) ON DELETE CASCADE,
  restaurant_id       UUID NOT NULL REFERENCES restaurants(id) ON DELETE CASCADE,
  name                TEXT NOT NULL,
  category            TEXT,
  by_glass_price      NUMERIC(10,2),
  bottle_price        NUMERIC(10,2),
  vintage             TEXT,
  region              TEXT,
  grape_variety       TEXT,
  wine_library_id     UUID REFERENCES master_wine_library(id) ON DELETE SET NULL,
  inventory_item_id   UUID,  -- FK to inventory table (added when inventory row is created)
  source              TEXT NOT NULL DEFAULT 'manual'
                      CHECK (source IN ('scan', 'csv', 'manual')),
  raw_extracted_text  TEXT,
  created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX IF NOT EXISTS idx_menu_items_menu ON menu_items (menu_id);
CREATE INDEX IF NOT EXISTS idx_menu_items_restaurant ON menu_items (restaurant_id);
ALTER TABLE menu_items ENABLE ROW LEVEL SECURITY;
CREATE POLICY "menu_items_restaurant_isolation" ON menu_items
  FOR ALL USING (restaurant_id = (SELECT restaurant_id FROM users WHERE id = auth.uid()));
```

### Task 3: user_onboarding_progress migration
```sql
CREATE TABLE IF NOT EXISTS user_onboarding_progress (
  id                      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id                 UUID NOT NULL UNIQUE REFERENCES users(id) ON DELETE CASCADE,
  restaurant_id           UUID NOT NULL REFERENCES restaurants(id) ON DELETE CASCADE,
  menu_uploaded           BOOLEAN NOT NULL DEFAULT FALSE,
  vendor_added            BOOLEAN NOT NULL DEFAULT FALSE,
  team_member_invited     BOOLEAN NOT NULL DEFAULT FALSE,
  completed_at            TIMESTAMPTZ,
  created_at              TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
ALTER TABLE user_onboarding_progress ENABLE ROW LEVEL SECURITY;
CREATE POLICY "onboarding_progress_self" ON user_onboarding_progress
  FOR ALL USING (user_id = auth.uid());
```

### Task 4: Update auth.service.ts to seed progress row
In `registerRestaurant()`, after creating the user and restaurant rows:
```typescript
await this.supabase
  .from('user_onboarding_progress')
  .insert({ user_id: userId, restaurant_id: restaurantId });
```
