class TestCompositeScore:

    def test_single_source_returns_none(self):
        """< 2 sources → None"""
        result = compute_composite_score({"wine_advocate": {"normalized_score": 93.0}})
        assert result is None

    def test_empty_scores_returns_none(self):
        assert compute_composite_score({}) is None

    def test_two_sources_returns_float(self):
        scores = {
            "wine_advocate": {"normalized_score": 93.0},
            "vivino": {"normalized_score": 84.0},
        }
        result = compute_composite_score(scores)
        assert result is not None
        assert isinstance(result, float)
        # WA(30%) × 93 + Vivino(20%) × 84 / (0.30 + 0.20) = (27.9 + 16.8) / 0.50 = 89.4
        assert abs(result - 89.4) < 0.5  # allow rounding tolerance

    def test_all_five_sources(self):
        scores = {
            "wine_advocate": {"normalized_score": 93.0},
            "wine_spectator": {"normalized_score": 91.0},
            "vivino": {"normalized_score": 84.0},
            "decanter": {"normalized_score": 95.0},
            "jancis_robinson": {"normalized_score": 82.5},
        }
        result = compute_composite_score(scores)
        assert result is not None
        assert 80.0 <= result <= 100.0

    def test_not_found_entries_ignored(self):
        """Sources with status=not_found have no normalized_score → not counted"""
        scores = {
            "wine_advocate": {"normalized_score": 93.0},
            "vivino": {"status": "not_found"},  # no normalized_score key
        }
        result = compute_composite_score(scores)
        assert result is None  # only 1 valid source

    def test_weights_sum_to_one(self):
        """D-05: locked weights must sum to 1.0"""
        total = sum(SCORE_WEIGHTS.values())
        assert abs(total - 1.0) < 0.001

import pytest

class TestRedisNXDedup:
    """CRIT-01 / D-03c: Redis SET NX dedup prevents double processing."""

    def test_second_call_returns_none_when_lock_held(self):
        """When lock is already held (NX set returns None), task returns None immediately."""
        from jobs.score_tasks import score_lookup_task

        mock_redis = MagicMock()
        mock_redis.set.return_value = None  # NX failed — lock already held
        mock_redis.delete.return_value = 1

        with patch("jobs.score_tasks.redis_lib") as mock_redis_lib:
            mock_redis_lib.from_url.return_value = mock_redis
            # Call task directly (not via .delay()) for synchronous test
            result = score_lookup_task.run("test-wine-id-123")

        assert result is None
        mock_redis.set.assert_called_once_with("wine:scores:test-wine-id-123", "1", nx=True, ex=3600)

    def test_lock_acquired_proceeds(self):
        """When lock is acquired (NX set returns True), task proceeds."""
        from jobs.score_tasks import score_lookup_task

        mock_redis = MagicMock()
        mock_redis.set.return_value = True  # NX succeeded — lock acquired

        with patch("jobs.score_tasks.redis_lib") as mock_redis_lib, \
             patch("jobs.score_tasks.asyncio") as mock_asyncio:
            mock_redis_lib.from_url.return_value = mock_redis
            mock_asyncio.run.return_value = {"wine_id": "test-id", "sources_found": 0}
            result = score_lookup_task.run("test-id")

        # Lock must be released in finally
        mock_redis.delete.assert_called_once()
        assert result == {"wine_id": "test-id", "sources_found": 0}

    def test_dataset_enrich_dedup_key_is_different(self):
        """dataset_enrich_task uses wine:dataset:{wine_id} NOT wine:scores:{wine_id}"""
