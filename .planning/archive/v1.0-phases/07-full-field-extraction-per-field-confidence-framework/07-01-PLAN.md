```python
"""
Field Confidence Module
=======================
Shared helpers for building, merging, and routing per-field confidence data.
Used by: claude_vision_extractor.py, haiku_enrichment_service.py,
         haiku_tasks.py, onboarding_routes.py, quality_routes.py
"""

from typing import Any, Dict, List, Optional, Tuple

# 18 fields Claude Vision is asked to extract (per D-04)
VISION_FIELDS = [
    "wine_name", "producer", "vintage", "primary_type", "color",
    "country", "region", "sub_region", "appellation", "grape_variety",
    "alcohol_pct", "price_bottle", "price_glass", "tasting_notes",
    "description", "section_name", "bin_number", "sweetness_level",
]

# Fields Haiku enrichment provides (per D-07)
ENRICHMENT_FIELDS = [
    "producer", "region", "sub_region", "appellation", "country",
    "grape_variety", "color", "primary_type", "sweetness_level",
    "food_pairing", "producer_bio", "tasting_notes", "alcohol_pct",
    "description",
]

# 6 JSONB enrichment keys (per D-07)
JSONB_ENRICHMENT_KEYS = [
    "grape_family", "wine_structure", "sensory_profile",
    "practical_attributes", "region_hierarchy", "critic_scores",
]

# Default thresholds (per D-04 / FCONF-04) — must match DB seed values
DEFAULT_REVIEW_THRESHOLD = 0.5
DEFAULT_ACCEPT_THRESHOLD = 0.8

# Auto-block threshold (per D-02): wine blocked when > 50% fields below reject threshold
AUTO_BLOCK_FIELD_RATIO = 0.5


def build_field_confidence(
    wine_dict: Dict[str, Any],
    source: str = "visible",
) -> Dict[str, Dict[str, Any]]:
    """
    Build field_confidence JSONB from a flat wine dict returned by Claude Vision.
    Each field in wine_dict that has a nested {value, confidence, source} structure
    is kept as-is. Flat values (legacy format) are wrapped with confidence=0.5, source=source.

    Returns: {"field_name": {"value": ..., "confidence": float, "source": str}, ...}
    """
    fc = {}
    for field_name in VISION_FIELDS:
        raw = wine_dict.get(field_name)
        if raw is None:
            continue
        if isinstance(raw, dict) and "confidence" in raw:
            fc[field_name] = {
                "value": raw.get("value"),
                "confidence": float(raw.get("confidence", 0.5)),
                "source": raw.get("source", source),
            }
        else:
            fc[field_name] = {
                "value": raw,
                "confidence": 0.5,
                "source": source,
            }
    return fc


def merge_field_confidence(
    existing_fc: Dict[str, Dict[str, Any]],
    new_fc: Dict[str, Dict[str, Any]],
    overwrite_lower: bool = True,
) -> Dict[str, Dict[str, Any]]:
    """
    Merge new field_confidence entries into existing.
    If overwrite_lower=True (default), new entry replaces existing only if
    existing field is absent OR new confidence >= existing confidence.
    This ensures Vision data is not overwritten by lower-confidence Haiku data
    for the same field.
    """
    merged = dict(existing_fc)
    for field_name, new_entry in new_fc.items():
        existing_entry = merged.get(field_name)
        if existing_entry is None:
            merged[field_name] = new_entry
        elif overwrite_lower and new_entry.get("confidence", 0) >= existing_entry.get("confidence", 0):
            merged[field_name] = new_entry
    return merged


def route_fields_by_threshold(
    fc: Dict[str, Dict[str, Any]],
    review_threshold: float = DEFAULT_REVIEW_THRESHOLD,
    accept_threshold: float = DEFAULT_ACCEPT_THRESHOLD,
) -> Tuple[Dict[str, Any], List[Dict[str, Any]], Dict[str, Any]]:
    """
    Route field_confidence entries through 3-tier threshold (per FCONF-04).

    Returns:
        accepted: dict of field_name -> value (confidence > accept_threshold)
        review:   list of dicts with field_name, current_value, confidence, source
                  (review_threshold <= confidence <= accept_threshold)
        rejected: dict of field_name -> None (confidence < review_threshold)
    """
    accepted = {}
    review = []
    rejected = {}

    for field_name, entry in fc.items():
        conf = entry.get("confidence", 0.0)
        value = entry.get("value")
        source = entry.get("source", "unknown")

        if conf > accept_threshold:
            accepted[field_name] = value
        elif conf >= review_threshold:
            accepted[field_name] = value  # Persisted but flagged
            review.append({
                "field_name": field_name,
                "current_value": str(value) if value is not None else None,
                "confidence": conf,
                "source": source,
            })
        else:
            rejected[field_name] = None

    return accepted, review, rejected


def compute_completeness_from_fc(
    fc: Dict[str, Dict[str, Any]],
    fields: Optional[List[str]] = None,
) -> float:
    """
    Compute completeness score from field_confidence JSONB (per D-01).
    Replaces the old compute_completeness() that counted non-null flat fields.
    Score = average confidence across all present fields.
    If fields param provided, only those fields are considered.
    """
    if not fc:
        return 0.0
    target_fields = fields or list(fc.keys())
    confidences = []
    for f in target_fields:
        entry = fc.get(f)
        if entry and entry.get("value") is not None:
            confidences.append(entry.get("confidence", 0.0))
        else:
            confidences.append(0.0)
    if not confidences:
        return 0.0
    return round(sum(confidences) / len(confidences), 3)


def should_auto_block(fc: Dict[str, Dict[str, Any]]) -> bool:
    """
    Per D-02: wine is auto_blocked when > 50% of its fields fall below
    the reject threshold (0.5). A wine with 2 bad fields out of 18 is NOT blocked.
    """
    if not fc:
        return True
    total = len(fc)
    if total == 0:
        return True
    rejected_count = sum(
        1 for entry in fc.values()
        if entry.get("confidence", 0.0) < DEFAULT_REVIEW_THRESHOLD
    )
    return (rejected_count / total) > AUTO_BLOCK_FIELD_RATIO
```
