"""
Field Confidence Calibration Task
==================================
Daily Celery beat task that measures actual accuracy per field per confidence bin,
then adjusts thresholds in confidence_thresholds table to maintain >= 0.95 accuracy.

Per D-21: Reads field_corrections + field_review_queue resolved entries.
Per D-22: Adjusts thresholds by +/- 0.05 based on accuracy, clamped to [0.3, 0.95].
Per D-23: Only calibrates fields with >= 50 total reviews (statistical significance).
"""

import logging
from datetime import datetime, timezone

from jobs.celery_app import celery_app
from config.settings import get_settings
from supabase import create_client

logger = logging.getLogger(__name__)
settings = get_settings()

MIN_REVIEWS_FOR_CALIBRATION = 50
ACCURACY_TARGET_LOW = 0.95   # below this -> raise accept_threshold
ACCURACY_TARGET_HIGH = 0.98  # above this -> lower review_threshold
THRESHOLD_STEP = 0.05
THRESHOLD_MIN = 0.30
THRESHOLD_MAX = 0.95

# 10 confidence bins
CONFIDENCE_BINS = [
    "0.0-0.1", "0.1-0.2", "0.2-0.3", "0.3-0.4", "0.4-0.5",
    "0.5-0.6", "0.6-0.7", "0.7-0.8", "0.8-0.9", "0.9-1.0",
]


def _get_supabase():
    return create_client(settings.supabase_url, settings.supabase_key)


def _confidence_to_bin(confidence: float) -> str:
    """Map a confidence value to its bin string."""
    idx = min(int(confidence * 10), 9)
    return CONFIDENCE_BINS[idx]


@celery_app.task(name="calibration.calibrate_field_thresholds")
def calibrate_field_thresholds_task():
    """
    Daily calibration task. Steps:
    1. Read all resolved field_review_queue entries (status in approved, corrected, rejected)
    2. For each field+bin, compute total_reviewed and total_correct
       - "approved" = correct (original value accepted)
       - "corrected" = incorrect (value was changed)
       - "rejected" = incorrect (value was thrown out)
    3. Write results to field_calibration table (upsert per field+bin)
    4. For fields with >= 50 reviews: adjust confidence_thresholds
    """
    supabase = _get_supabase()
    now_iso = datetime.now(timezone.utc).isoformat()

    try:
        # 1. Read resolved field_review_queue entries
        resp = (
            supabase.table("field_review_queue")
            .select("field_name, confidence, status")
            .in_("status", ["approved", "corrected", "rejected"])
            .execute()
        )
        resolved = resp.data or []
    except Exception as exc:
        logger.error("Calibration: failed to read field_review_queue: %s", exc)
        return {"error": str(exc)}

    if not resolved:
        logger.info("Calibration: no resolved reviews found, skipping")
        return {"status": "skipped", "reason": "no_resolved_reviews"}

    # 2. Aggregate per field + bin
    stats = {}  # (field_name, bin) -> {total: int, correct: int}
    for row in resolved:
        fname = row["field_name"]
        conf = float(row["confidence"])
        bin_str = _confidence_to_bin(conf)
        key = (fname, bin_str)
        if key not in stats:
            stats[key] = {"total": 0, "correct": 0}
        stats[key]["total"] += 1
        if row["status"] == "approved":
            stats[key]["correct"] += 1
        # corrected and rejected are NOT correct

    # 3. Upsert field_calibration rows
    calibration_rows = []
    for (fname, bin_str), s in stats.items():
        accuracy = s["correct"] / s["total"] if s["total"] > 0 else 0.0
        calibration_rows.append({
            "field_name": fname,
            "confidence_bin": bin_str,
            "total_reviewed": s["total"],
            "total_correct": s["correct"],
            "actual_accuracy": round(accuracy, 4),
            "measured_at": now_iso,
        })

    try:
        supabase.table("field_calibration").upsert(
            calibration_rows,
            on_conflict="field_name,confidence_bin",
        ).execute()
        logger.info("Calibration: wrote %d field_calibration rows", len(calibration_rows))
    except Exception as exc:
        logger.error("Calibration: field_calibration upsert failed: %s", exc)

    # 4. Adjust thresholds for fields with sufficient data
    field_totals = {}  # field_name -> total_reviewed across all bins
    for (fname, _), s in stats.items():
        field_totals[fname] = field_totals.get(fname, 0) + s["total"]

    adjusted = 0
    for fname, total in field_totals.items():
        if total < MIN_REVIEWS_FOR_CALIBRATION:
            continue

        # Get current thresholds
        try:
            t_resp = (
                supabase.table("confidence_thresholds")
                .select("review_threshold, accept_threshold")
                .eq("field_name", fname)
                .maybe_single()
                .execute()
            )
            if not t_resp.data:
                continue
            current_review = float(t_resp.data["review_threshold"])
            current_accept = float(t_resp.data["accept_threshold"])
        except Exception:
            continue

        new_review = current_review
        new_accept = current_accept

        # Check accuracy in bins around accept_threshold
        for (fn, bin_str), s in stats.items():
            if fn != fname or s["total"] < 5:
                continue
            accuracy = s["correct"] / s["total"]
            # If accuracy in any bin < 0.95, raise accept_threshold
            if accuracy < ACCURACY_TARGET_LOW:
                new_accept = min(current_accept + THRESHOLD_STEP, THRESHOLD_MAX)
            # If accuracy in high bins > 0.98, lower review_threshold
            if accuracy > ACCURACY_TARGET_HIGH:
                new_review = max(current_review - THRESHOLD_STEP, THRESHOLD_MIN)

        if new_review != current_review or new_accept != current_accept:
            try:
                supabase.table("confidence_thresholds").update({
                    "review_threshold": new_review,
                    "accept_threshold": new_accept,
                    "last_calibrated_at": now_iso,
                }).eq("field_name", fname).execute()
                adjusted += 1
                logger.info(
                    "Calibration: adjusted %s thresholds: review=%.2f->%.2f, accept=%.2f->%.2f",
                    fname, current_review, new_review, current_accept, new_accept,
                )
            except Exception as exc:
                logger.warning("Calibration: threshold update failed for %s: %s", fname, exc)

    return {
        "status": "completed",
        "resolved_reviews": len(resolved),
        "calibration_rows": len(calibration_rows),
        "thresholds_adjusted": adjusted,
    }

@router.get("/calibration")
def get_calibration():
    """
    GET /api/v1/quality/calibration
    Returns current per-field thresholds, accuracy stats per bin, sample sizes.
    Dashboard-ready JSON.
    """
    supabase = _get_supabase()
    if not supabase:
        raise HTTPException(status_code=503, detail="Database not available")

    try:
        thresholds_resp = (
            supabase.table("confidence_thresholds")
            .select("field_name, review_threshold, accept_threshold, last_calibrated_at")
            .order("field_name")
            .execute()
        )
        calibration_resp = (
            supabase.table("field_calibration")
            .select("field_name, confidence_bin, total_reviewed, total_correct, actual_accuracy, measured_at")
            .order("field_name")
            .execute()
        )
    except Exception as exc:
        logger.error("get_calibration DB error: %s", exc)
        raise HTTPException(status_code=503, detail="Database query failed")

    # Group calibration by field
    cal_by_field = {}
    for row in (calibration_resp.data or []):
        fname = row["field_name"]
        if fname not in cal_by_field:
            cal_by_field[fname] = []
        cal_by_field[fname].append({
            "bin": row["confidence_bin"],
            "total_reviewed": row["total_reviewed"],
            "total_correct": row["total_correct"],
            "accuracy": float(row["actual_accuracy"]) if row["actual_accuracy"] else None,
        })

    return {
        "thresholds": thresholds_resp.data or [],
        "calibration_data": cal_by_field,
        "total_fields_tracked": len(thresholds_resp.data or []),
    }
