**Script structure:**
```python
#!/usr/bin/env python3
"""
generate_ontology_seed.py
=========================
Calls Claude Opus to generate comprehensive wine ontology seed data.
Outputs 4 SQL seed files to supabase/migrations/seed/.

Usage:
    ANTHROPIC_API_KEY=... python3 scripts/generate_ontology_seed.py

Cost estimate: ~$2-5 total (claude-opus-4-5-20251001, 8 API calls, ~2000 output tokens each)
"""
import os
import sys
import json
import pathlib
import anthropic

SEED_DIR = pathlib.Path("supabase/migrations/seed")
SEED_DIR.mkdir(parents=True, exist_ok=True)

MODEL = "claude-opus-4-5-20251001"  # Verify this is current; if invalid, try "claude-opus-4-0" or check Anthropic API docs
CLIENT = anthropic.Anthropic(api_key=os.environ["ANTHROPIC_API_KEY"])

def call_claude(prompt: str, max_tokens: int = 8096) -> str:
    """Call Claude Opus and return the text content."""
    msg = CLIENT.messages.create(
        model=MODEL,
        max_tokens=max_tokens,
        messages=[{"role": "user", "content": prompt}],
    )
    return msg.content[0].text
```

**Script writes each batch to files, then concatenates:**
```python
def write_seed_file(filename: str, header: str, sql_batches: list[str]) -> int:
    """Write combined INSERT statements to seed file. Returns row count."""
    path = SEED_DIR / filename
    lines = []
    for batch in sql_batches:
        for line in batch.strip().split("\n"):
            line = line.strip()
            if line.startswith("INSERT INTO") and line.endswith(";"):
                lines.append(line)
    path.write_text(f"-- {header}\n-- Generated by scripts/generate_ontology_seed.py\n-- Source: Claude Opus structured prompt\n\n" + "\n".join(lines) + "\n")
    return len(lines)
```
