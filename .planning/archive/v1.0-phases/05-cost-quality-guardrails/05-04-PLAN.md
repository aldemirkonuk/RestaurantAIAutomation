**Create services/agent-orchestrator/api/quality_routes.py:**

```python
"""
Quality Routes
==============
Human review queue: GET surfaces wines needing review, PATCH accepts corrections
and logs per-field deltas to field_corrections table for acceptance rate tracking.

GET  /api/v1/quality/review-queue
PATCH /api/v1/quality/review-queue/{submission_id}
"""

import logging
from datetime import datetime, timezone
from typing import Any, Dict, List, Optional

from fastapi import APIRouter, HTTPException
from pydantic import BaseModel

logger = logging.getLogger(__name__)

router = APIRouter(prefix="/api/v1/quality", tags=["quality"])

# Fields used to compute completeness_score (mirrors claude_vision_extractor.py SCORED_FIELDS)
SCORED_FIELDS = [
    "wine_name", "vintage", "price_bottle", "region",
    "country", "grape_variety", "section_name",
]
COMPLETENESS_THRESHOLD = 0.5  # below this → needs_review; auto_blocked threshold is 0.3


# =============================================================================
# HELPERS
# =============================================================================

def get_supabase_client():
    """Return Supabase client from settings. Returns None if not configured."""
    try:
        from config.settings import get_settings
        settings = get_settings()
        return getattr(settings, "supabase_client", None) or getattr(settings, "supabase", None)
    except Exception:
        return None


def _compute_completeness(wine: dict) -> float:
    """Recompute completeness score from current field values."""
    if not wine:
        return 0.0
    filled = sum(1 for f in SCORED_FIELDS if wine.get(f) not in (None, "", []))
    return round(filled / len(SCORED_FIELDS), 4)


# =============================================================================
# REQUEST / RESPONSE MODELS
# =============================================================================

class WineCorrectionRequest(BaseModel):
    """
    Fields the human reviewer is correcting.
    Only include fields being changed — others are left as-is.
    """
    wine_name: Optional[str] = None
    vintage: Optional[str] = None
    price_bottle: Optional[str] = None
    price_glass: Optional[str] = None
    region: Optional[str] = None
    country: Optional[str] = None
    grape_variety: Optional[str] = None
    section_name: Optional[str] = None
    bin_number: Optional[str] = None
    corrected_by: Optional[str] = None  # reviewer email or user id


# =============================================================================
# ENDPOINTS
# =============================================================================

@router.get("/review-queue")
async def get_review_queue(limit: int = 100):
    """
    GET /api/v1/quality/review-queue

    Returns wines from master_wine_library_submissions where:
      needs_review = true  (completeness 0.3–0.5)
      OR auto_blocked = true  (completeness < 0.3)

    Ordered by completeness_score ASC (worst wines first for reviewers).
    """
    supabase = get_supabase_client()
    if supabase is None:
        raise HTTPException(status_code=503, detail="Supabase not configured")

    resp = (
        supabase.table("master_wine_library_submissions")
        .select("*")
        .or_("needs_review.eq.true,auto_blocked.eq.true")
        .order("completeness_score", desc=False)
        .limit(limit)
        .execute()
    )
    items = resp.data or []
    return {"items": items, "count": len(items)}


@router.patch("/review-queue/{submission_id}")
async def patch_review_queue(submission_id: str, correction: WineCorrectionRequest):
    """
    PATCH /api/v1/quality/review-queue/{submission_id}

    Accepts human corrections for a submission:
    1. Loads existing submission row
    2. Applies corrections to payload
    3. Recomputes completeness_score
    4. Logs each changed field to field_corrections (QUAL-02)
    5. If auto_blocked=True AND new completeness >= 0.5: promotes to master_wine_library

    Returns updated submission data.
    """
    supabase = get_supabase_client()
    if supabase is None:
        raise HTTPException(status_code=503, detail="Supabase not configured")

    # Step 1: Load existing submission row
    existing_resp = (
        supabase.table("master_wine_library_submissions")
        .select("*")
        .eq("id", submission_id)
        .execute()
    )
    if not existing_resp.data:
        raise HTTPException(status_code=404, detail=f"Submission {submission_id} not found")

    existing = existing_resp.data[0]
    existing_payload: dict = existing.get("payload") or {}
    was_auto_blocked: bool = existing.get("auto_blocked", False)

    # Step 2: Apply corrections — only non-None values from request
    correction_dict = {
        k: v for k, v in correction.model_dump().items()
        if v is not None and k != "corrected_by"
    }
    updated_payload = {**existing_payload, **correction_dict}

    # Step 3: Recompute completeness_score
    new_completeness = _compute_completeness(updated_payload)
    new_needs_review = new_completeness < COMPLETENESS_THRESHOLD
    new_auto_blocked = new_completeness < 0.3 and was_auto_blocked  # stays blocked unless threshold met

    # Step 4: Log each changed field to field_corrections (QUAL-02)
    corrected_at = datetime.now(timezone.utc).isoformat()
    corrected_by = correction.corrected_by
    corrections_to_log = []
    for field_name, new_value in correction_dict.items():
        original_value = existing_payload.get(field_name)
        if str(original_value) != str(new_value):  # only log genuine changes
            corrections_to_log.append({
                "submission_id": submission_id,
                "field_name": field_name,
                "original_value": str(original_value) if original_value is not None else None,
                "corrected_value": str(new_value),
                "corrected_at": corrected_at,
                "corrected_by": corrected_by,
            })

    if corrections_to_log:
        try:
            supabase.table("field_corrections").insert(corrections_to_log).execute()
        except Exception as exc:
            logger.warning(f"Failed to log field_corrections for {submission_id}: {exc}")

    # Step 5: Update submission row
    update_data: Dict[str, Any] = {
        "payload": updated_payload,
        "completeness_score": new_completeness,
        "needs_review": new_needs_review,
        "auto_blocked": new_auto_blocked,
    }
    if not new_needs_review and not new_auto_blocked:
        update_data["status"] = "approved"
    elif new_auto_blocked:
        update_data["status"] = "auto_blocked"
    else:
        update_data["status"] = "pending_review"

    supabase.table("master_wine_library_submissions").update(update_data).eq("id", submission_id).execute()

    # Step 5b: Promote to master_wine_library if auto_blocked was True and now completeness >= 0.5
    promoted = False
    if was_auto_blocked and new_completeness >= COMPLETENESS_THRESHOLD:
        try:
            promote_payload = {
                **updated_payload,
                "completeness_score": new_completeness,
                "promoted_from_submission_id": submission_id,
                "promotion_source": "human_review",
            }
            # Remove internal fields not needed in master library
            for internal_key in ("needs_review", "auto_blocked", "scan_session_id", "extraction_source", "extraction_cost_usd"):
                promote_payload.pop(internal_key, None)

            supabase.table("master_wine_library").upsert(
                promote_payload,
                on_conflict="id",
            ).execute()

            # Mark submission as promoted
            supabase.table("master_wine_library_submissions").update({
                "status": "promoted",
                "auto_blocked": False,
            }).eq("id", submission_id).execute()

            promoted = True
            logger.info(
                f"Auto-blocked wine promoted to master_wine_library: "
                f"submission_id={submission_id}, new_completeness={new_completeness}"
            )
        except Exception as exc:
            logger.error(f"Promotion to master_wine_library failed for {submission_id}: {exc}")

    return {
        "submission_id": submission_id,
        "completeness_score": new_completeness,
        "needs_review": new_needs_review,
        "auto_blocked": new_auto_blocked,
        "promoted_to_master_library": promoted,
        "fields_corrected": len(corrections_to_log),
        "corrections": corrections_to_log,
    }
```
