**Step 1 — Create jobs/spend_tasks.py:**

```python
"""
Spend Tasks — Monthly Cap Check Celery Beat Task
=================================================
Runs hourly via Celery beat. Queries api_spend SUM(cost_usd) per provider
for the current calendar month. Sends email alert when >= 80% of cap is reached.

Caps (COST-02): $50 Claude, $20 Gemini. Alert threshold: 80% ($40 / $16).
Idempotent: one alert per provider per calendar month (tracked in spend_alert_state).
"""

import asyncio
import logging
from datetime import datetime, timezone
from typing import Optional

from supabase import create_client

from config.settings import get_settings
from jobs.celery_app import celery_app

logger = logging.getLogger(__name__)

# Monthly caps and alert thresholds (COST-02)
MONTHLY_CAPS = {
    "anthropic": 50.0,
    "google": 20.0,
}
ALERT_THRESHOLD_PCT = 0.80  # alert at 80%
ALERT_THRESHOLDS = {
    provider: cap * ALERT_THRESHOLD_PCT
    for provider, cap in MONTHLY_CAPS.items()
}


def _get_supabase():
    settings = get_settings()
    return create_client(settings.supabase_url, settings.supabase_key)


def _get_month_start() -> str:
    now = datetime.now(timezone.utc)
    return datetime(now.year, now.month, 1, tzinfo=timezone.utc).isoformat()


def _current_month_str() -> str:
    """Returns 'YYYY-MM' string for current UTC month."""
    now = datetime.now(timezone.utc)
    return f"{now.year}-{now.month:02d}"


@celery_app.task(name="spend.monthly_cap_check")
def monthly_cap_check_task() -> dict:
    """
    Hourly Celery beat task. Checks monthly API spend against soft caps.
    Sends email alert when a provider crosses 80% of its monthly cap.
    Idempotent: only one alert per provider per calendar month.
    """
    return asyncio.run(_monthly_cap_check_async())


async def _monthly_cap_check_async() -> dict:
    """Async implementation of monthly cap check."""
    settings = get_settings()
    if not settings.supabase_url or not settings.supabase_key:
        logger.debug("spend_tasks: Supabase not configured — skipping cap check")
        return {"skipped": True, "reason": "supabase_not_configured"}

    supabase = _get_supabase()
    month_start = _get_month_start()
    current_month = _current_month_str()
    result = {"alerts_sent": [], "providers_checked": []}

    for provider, threshold in ALERT_THRESHOLDS.items():
        try:
            # Query monthly spend sum for this provider
            resp = (
                supabase.table("api_spend")
                .select("cost_usd")
                .eq("provider", provider)
                .gte("timestamp", month_start)
                .execute()
            )
            monthly_spend = sum(row["cost_usd"] for row in (resp.data or []))
            result["providers_checked"].append({
                "provider": provider,
                "monthly_spend": monthly_spend,
                "threshold": threshold,
                "cap": MONTHLY_CAPS[provider],
            })

            if monthly_spend < threshold:
                continue

            # Check idempotency: has alert already been sent this month?
            state_resp = (
                supabase.table("spend_alert_state")
                .select("last_alert_month")
                .eq("provider", provider)
                .execute()
            )
            state_rows = state_resp.data or []
            if state_rows and state_rows[0].get("last_alert_month") == current_month:
                logger.info(
                    f"Monthly cap alert already sent for {provider} in {current_month} — skipping"
                )
                continue

            # Send alert email
            if settings.manager_email and settings.gmail_user and settings.gmail_password:
                from services.email_client import EmailClient
                email_client = EmailClient(
                    backend="gmail",
                    gmail_user=settings.gmail_user,
                    gmail_password=settings.gmail_password,
                )
                cap = MONTHLY_CAPS[provider]
                pct = (monthly_spend / cap) * 100
                await email_client.send_email(
                    to=settings.manager_email,
                    subject=f"[WineOps] {provider.capitalize()} API spend alert: ${monthly_spend:.2f} ({pct:.0f}% of ${cap:.0f} cap)",
                    body=(
                        f"Monthly API spend alert for provider: {provider}\n\n"
                        f"Current spend this month: ${monthly_spend:.4f}\n"
                        f"Alert threshold (80%): ${threshold:.2f}\n"
                        f"Monthly cap: ${cap:.2f}\n\n"
                        f"Action: Review usage or adjust extraction frequency."
                    ),
                )

            # Upsert idempotency state
            supabase.table("spend_alert_state").upsert(
                {"provider": provider, "last_alert_month": current_month, "updated_at": datetime.now(timezone.utc).isoformat()},
                on_conflict="provider",
            ).execute()

            result["alerts_sent"].append(provider)
            logger.info(f"Monthly cap alert sent for {provider}: ${monthly_spend:.4f} spend")

        except Exception as exc:
            logger.error(f"spend_tasks: cap check failed for {provider}: {exc}")

    return result
```
